55def _build_header() -> html.Header: 

56 """v2-aligned header: monogram + wordmark + region/persona/mode controls.""" 

57 persona_options = [ 

58 {"label": f"{p['avatar']} {p['title']}", "value": p["id"]} for p in list_personas() 

59 ] 

60 # Region options are grouped geographically (Central / Northeast / 

61 # Southeast / West). dbc.Select doesn't support native <optgroup>, 

62 # so we surface each group with a disabled separator option that's 

63 # visually distinct but unselectable. The empty value="" never 

64 # reaches a callback because Bootstrap honors the disabled attribute 

65 # on click + keyboard nav. 

66 # 

67 # V3.ζ follow-up: the forecast quality gate 

68 # (``is_forecast_quality_acceptable``) hides a BA only when its best 

69 # served-model MAPE — the champion across the ensemble + 3 bases, not 

70 # XGBoost-alone (#255) — is in the ``rollback`` grade (>22% on 7d horizon). 

71 # Empty groups (e.g. all members hidden) drop their separator too. 

72 from models.model_service import is_forecast_quality_acceptable 

73 

74 region_options: list[dict] = [] 

75 for group_name, codes in REGION_GROUPS.items(): 

76 visible_codes = [c for c in codes if is_forecast_quality_acceptable(c)] 

77 if not visible_codes: 

78 continue 

79 region_options.append({"label": f"── {group_name} ──", "value": "", "disabled": True}) 

80 for code in visible_codes: 

81 region_options.append({"label": REGION_NAMES.get(code, code), "value": code}) 

82 

83 brand = html.Div( 

84 [ 

85 _monogram(), 

86 html.Div( 

87 [ 

88 html.H1( 

89 [ 

90 html.Span("Grid", className="gp-header__wordmark-grid"), 

91 html.Span("Pulse", className="gp-header__wordmark-pulse"), 

92 ], 

93 className="gp-header__wordmark dashboard-title", 

94 ), 

95 html.P( 

96 "Energy intelligence platform", 

97 className="gp-brand-tagline", 

98 ), 

99 ], 

100 className="gp-header__brand-lockup", 

101 ), 

102 ], 

103 className="gp-header__brand", 

104 ) 

105 

106 controls = html.Div( 

107 [ 

108 # dbc.Select accepts no aria-label/title prop, so the accessible name 

109 # comes from a visually-hidden <label htmlFor> (#224). 

110 html.Label("Region", htmlFor="region-selector", className="sr-only"), 

111 dbc.Select( 

112 id="region-selector", 

113 options=region_options, 

114 value="FPL", 

115 className="gp-header__select region-selector", 

116 ), 

117 html.Label("View", htmlFor="persona-selector", className="sr-only"), 

118 dbc.Select( 

119 id="persona-selector", 

120 options=persona_options, 

121 value="grid_ops", 

122 className="gp-header__chip persona-switcher", 

123 ), 

124 html.Button( 

125 "Briefing Mode", 

126 id="meeting-mode-btn", 

127 n_clicks=0, 

128 className="gp-header__link", 

129 ), 

130 html.Button( 

131 "Save View", 

132 id="bookmark-btn", 

133 n_clicks=0, 

134 className="gp-header__link", 

135 ), 

136 # Hidden — kept for the freshness callback that writes here. 

137 html.Div(id="header-freshness", style={"display": "none"}), 

138 ], 

139 className="gp-header__controls", 

140 ) 

141 

142 return html.Header( 

143 html.Div([brand, controls], className="gp-header__inner"), 

144 className="dashboard-header gp-header", 

145 id="dashboard-header", 

146 )

149def _tab(tab_id: str, layout_fn) -> dbc.Tab: 

150 """Wrap a tab module's layout in ``dbc.Tab``.""" 

151 label = _VISIBLE_LABEL_OVERRIDES.get(tab_id, TAB_LABELS.get(tab_id, tab_id)) 

152 return dbc.Tab(layout_fn(), label=label, tab_id=tab_id)

155def build_layout() -> dbc.Container: 

156 """Build the full dashboard layout (R3 — 4 visible tabs).""" 

157 return dbc.Container( 

158 [ 

159 # URL state for bookmarks (C2) 

160 dcc.Location(id="url", refresh=False),

163 # Header 

164 _build_header(), 

165 # Bookmark / save-view toast 

166 html.Div( 

167 id="bookmark-toast", 

168 style={ 

169 "position": "fixed", 

170 "top": "10px", 

171 "right": "10px", 

172 "zIndex": 9999, 

173 }, 

174 ), 

175 # Data freshness banner (G2 — visible only when degraded) 

176 html.Div(id="fallback-banner"), 

177 # Per-Widget confidence badges (A4 + E3) — hidden carrier element 

178 html.Div(id="widget-confidence-bar", style={"display": "none"}), 

179 # Tab strip (4 visible: Overview / Forecast / Risk / Models) 

180 # R5b: wrapped in <main role="main"> so the skip-to-content

182 # main content region. The dbc.Tabs strip itself is the 

183 # primary navigation control.

185 dbc.Tabs(

194 ], 

195 ), 

196 id="main-content", 

197 role="main", 

198 ),
