1"""Alerts / Extreme Events tab helpers extracted from ``components/callbacks.py``. 

2 

3Step 4 of the ``callbacks.py`` decomposition tracked in issue #87. 

4Continues the per-tab split established by: 

5 

6* #98 — shared infrastructure (``_callbacks_shared.py``) 

7* #99 — US Grid tab (``_callbacks_us_grid.py``) 

8* #100 — Models tab (``_callbacks_models.py``) 

9 

10## What lives here 

11 

12``_render_risk_tab`` plus the per-widget builders it composes, and the 

13two *data* paths that feed it: 

14 

15* ``_alerts_tab_from_redis`` — production. Reads the 

16 ``gridpulse:alerts:{region}`` payload the scoring job writes hourly. 

17* the dev/demo fallback inside ``update_alerts_tab`` — computes the same 

18 shapes locally from ``demand-store`` / ``weather-store`` when 

19 ``REQUIRE_REDIS`` is off. 

20 

21**One renderer, two data sources (P2-44 / #273).** These two paths used 

22to build the tab independently and had drifted into *different designs*: 

23production rendered the stress breakdown as emoji plus inline styles 

24while dev rendered the ``gp-stress-row`` icon components, and the dev 

25path printed a literal ``"None"`` for BAs whose capacity is unreliable. 

26The split that remains is the honest one — where the numbers come from — 

27and the rendering below it is shared, so the two can no longer diverge. 

28 

29The docstring previously claimed this module held "a single function" 

30with "no fallback compute path"; the dev fallback has computed grid 

31stress inline since #265. Corrected here. 

32 

33When the alerts tab grows additional helpers (severity classifiers, 

34event-timeline filters, etc.), they belong here. 

35

41 

42When patching for tests, target the function's *new* namespace:

71 """kpi-delta tone for the utilization-based grid-stress score (#265). 

72

74 a missing score is neutral. Bands mirror ``models.pricing.grid_stress``. 

75 """ 

76 if stress is None: 

77 return "neutral" 

78 return "positive" if stress < 70 else ("negative" if stress >= 85 else "neutral")

82#: (P2-29 / #273). They are drawn identically for all 51 BAs, so 95°F reads as 

83#: a meaningful mark in Florida and as an unreachable one in the Pacific 

84#: Northwest. The chart now labels them as generic rather than implying each BA

236def _read_temp_percentiles(region: str) -> dict | None: 

237 """Per-BA temperature percentiles from Redis, or ``None`` (#401). 

238 

239 Redis-only, per the web-tier I/O guardrail — the scoring job is the sole 

240 writer. A miss returns ``None`` and the caller falls back to the generic 

241 lines *and* their generic disclosure, so a cold cache degrades to the 

242 previous behaviour rather than to a blank chart or a fabricated threshold. 

243 """ 

244 try: 

245 payload = redis_get(redis_key(f"weather_percentiles:{region}")) 

246 except Exception: # pragma: no cover — advisory; never break the tab 

247 return None 

248 if not isinstance(payload, dict): 

249 return None 

250 if not all(isinstance(payload.get(k), (int, float)) for k in ("p90", "p95", "p99")): 

251 return None 

252 return payload

255def _build_temp_figure(region, timestamps, values): 

256 """Temperature series with heat reference lines. 

257 

258 Region-calibrated when the scoring job has published percentiles for this 

259 BA (#401), generic and labelled as such otherwise (#273). The label always 

260 states which it is and, when calibrated, over what window — a reader must 

261 never have to guess whether a line was assessed for this BA. 

262 """ 

263 if len(timestamps) == 0: 

264 return _empty_figure("Loading...") 

265 fig = go.Figure() 

266 fig.add_trace( 

267 go.Scatter( 

268 x=timestamps, y=values, name="Temperature", line=dict(color=COLORS["temperature"]) 

269 ) 

270 ) 

271 

272 pct = _read_temp_percentiles(region) 

273 if pct: 

274 lines = [("p90", pct["p90"]), ("p95", pct["p95"]), ("p99", pct["p99"])] 

275 days = pct.get("window_days") 

276 title = f"Calibrated to {region} — p90/p95/p99 of observed temperature" + ( 

277 f", trailing {days} days" if days else "" 

278 ) 

279 else: 

280 lines = [(f"{t}\u00b0F", t) for t in _TEMP_REFERENCE_LINES_F] 

281 title = "Generic heat reference levels — not calibrated per region" 

282 

283 for label, t in lines: 

284 fig.add_hline( 

285 y=t, 

286 line=dict(color="#FF5C7A", dash="dot", width=1), 

287 annotation_text=f"{label} \u00b7 {t:.0f}\u00b0F" if pct else label, 

288 annotation_position="right", 

289 ) 

290 fig.update_layout( 

291 **_layout( 

292 uirevision=region, 

293 yaxis_title="\u00b0F", 

294 title=dict( 

295 text=title, 

296 font=dict(size=10, color="#A8B3C7"), 

297 x=0, 

298 xanchor="left", 

299 y=0.99, 

300 yanchor="top", 

301 ), 

302 ) 

303 ) 

304 return fig

307def _build_timeline_figure(region): 

308 """Categorical timeline of historical extreme events. 

309 

310 Events in ``region`` are highlighted. There is deliberately no y-axis: 

311 these events used to be plotted against invented 0–100 "severity scores" 

312 (P2-29 / #273), which rendered editorial judgement as measurement on a 

313 quantitative axis. The dates and regions are factual; the scores were not, 

314 so they are gone rather than restated. 

315 """ 

316 fig = go.Figure() 

317 for date, name, reg in _HISTORICAL_EVENTS: 

318 in_region = reg == region 

319 fig.add_trace( 

320 go.Scatter( 

321 x=[date], 

322 y=[0], 

323 mode="markers+text", 

324 text=[f"{name}<br><span style='font-size:9px'>{reg}</span>"], 

325 textposition="top center", 

326 marker=dict( 

327 size=14 if in_region else 10, 

328 color=COLORS["ensemble"] if in_region else "#A8B3C7", 

329 symbol="diamond" if in_region else "circle", 

330 ), 

331 showlegend=False, 

332 hovertemplate=f"{name} · {reg}<br>%{{x}}<extra></extra>", 

333 ) 

334 ) 

335 # Pad the date axis so the first and last event labels are not clipped 

336 # against the plot edges — the markers sit at the extremes of the data 

337 # range, and the labels are centred above them. 

338 dates = sorted(pd.to_datetime([e[0] for e in _HISTORICAL_EVENTS])) 

339 pad = pd.Timedelta(days=240) 

340 fig.update_layout( 

341 **_layout( 

342 uirevision=region, 

343 xaxis_title="Date", 

344 xaxis=dict(range=[(dates[0] - pad).isoformat(), (dates[-1] + pad).isoformat()]), 

345 yaxis=dict(visible=False, range=[-1, 1.6]), 

346 ) 

347 ) 

348 return fig

397def _alerts_tab_from_redis(region): 

398 """Redis fast path for update_alerts_tab callback. 

399 

400 Returns an 8-tuple (alert_cards, stress_str, stress_label_span, breakdown, 

401 fig_anomaly, fig_temp, fig_timeline, weather_context) or None if cache miss. 

402 """ 

403 cached = redis_get(redis_key(f"alerts:{region}")) 

404 if cached is None: 

405 return None 

406 

407 log.info("alerts_redis_hit", region=region) 

408 alerts = cached.get("alerts", []) 

409 # Legacy payloads (pre-alerts_source) only ever carried demo content. 

410 alerts_source = cached.get("alerts_source", "demo") 

411 stress = cached.get("stress_score") 

412 stress_label = cached.get("stress_label", "Unavailable") 

413 counts = cached.get("alert_counts", {}) 

414 anomaly = dict(cached.get("anomaly", {})) 

415 temp_data = cached.get("temperature", {}) 

416 t_vals = temp_data.get("values", []) 

417 anomaly["timestamps"] = pd.to_datetime(anomaly.get("timestamps", [])) 

418 anomaly["anomaly_timestamps"] = pd.to_datetime(anomaly.get("anomaly_timestamps", [])) 

419 

420 # Cards, breakdown and figures are built by the shared renderer below; 

421 # this path's only job is resolving the data out of the payload. 

422 # Weather context — the full "Current Conditions" card row (temperature + 

423 # wind + humidity + cloud), built from the scoring job's latest reading. The 

424 # web tier used to only have the temperature series, so it rendered a lone 

425 # Temperature card; the scoring job now ships ``weather_current`` so this 

426 # matches the dev path's multi-card view. 

427 weather_context = html.Div() 

428 weather_current = cached.get("weather_current") 

429 if weather_current: 

430 weather_context = _build_weather_context(pd.Series(weather_current)) 

431 elif t_vals: 

432 # Back-compat: older payloads (pre this change, or Redis not yet 

433 # refreshed) carry only the temperature series — show that one card. 

434 import dash_bootstrap_components as dbc 

435 

436 last_temp = float(t_vals[-1]) 

437 if pd.isna(last_temp): 

438 temp_display = "—" 

439 color = "#A8B3C7" 

440 else: 

441 temp_display = f"{last_temp:.0f}°F" 

442 color = "#FF5C7A" if last_temp >= 95 else ("#FFB84D" if last_temp >= 85 else "#2BD67B") 

443 weather_context = dbc.Row( 

444 [ 

445 dbc.Col( 

446 html.Div( 

447 [ 

448 html.P("TEMPERATURE", className="kpi-label"), 

449 html.H4( 

450 temp_display, 

451 className="kpi-value", 

452 style={"fontSize": "1.3rem"}, 

453 ), 

454 ], 

455 className="kpi-card", 

456 style={"borderTop": f"3px solid {color}"}, 

457 ), 

458 md=3, 

459 ), 

460 ], 

461 className="g-2", 

462 ) 

463 

464 return _render_risk_tab( 

465 region=region, 

466 alerts=alerts, 

467 alerts_source=alerts_source, 

468 alerts_total=cached.get("alerts_total"), 

469 stress=stress, 

470 stress_label=stress_label, 

471 counts=counts, 

472 anomaly=anomaly, 

473 temperature=temp_data, 

474 weather_context=weather_context, 

475 )

481def register_alerts_callbacks(app): 

482 """Register Risk / Alerts tab callbacks with the Dash app. 

483 

484 Step 10d of the ``register_callbacks`` decomposition (issue #87). 

485 Owns the three callbacks driving the Risk tab: 

486 

487 * ``update_risk_title`` — page title block. 

488 * ``update_risk_insight`` — 1-sentence narrative summary 

489 (``'all systems nominal'`` or an elevated-risk note). 

490 * ``update_alerts_tab`` — 8-output payload: alert cards, stress 

491 score / label / breakdown, anomaly chart, temperature exceedance, 

492 historical event timeline, weather-context KPIs. Has Redis fast 

493 path (``_alerts_tab_from_redis``) and a v1 demo-data compute 

494 fallback for dev mode. 

495 

496 Section comment in callbacks.py was historically labeled 

497 ``# ── 7. TAB 4: GENERATION & NET LOAD ──`` even though every 

498 output id is ``tab5-*`` (the Risk tab). The label survived an 

499 earlier round of refactoring that deleted the Generation TAB 4 

500 callbacks but left the breadcrumb. Corrected when this section 

501 moved into ``register_alerts_callbacks(app)``. 

502 """ 

503 from components.cards import build_page_title 

504 from config import REGION_NAMES 

505 

506 @app.callback( 

507 Output("risk-title", "children"), 

508 [ 

509 Input("region-selector", "value"), 

510 Input("dashboard-tabs", "active_tab"), 

511 ],

514 """Page-title block for the Risk tab.""" 

515 if active_tab != "tab-alerts": 

516 return no_update 

517

519 region_name = REGION_NAMES.get(region, region) 

520 return build_page_title( 

521 "Risk", 

522 f"Active alerts, demand anomalies, and grid stress · {region_name}", 

523 ) 

524 

525 @app.callback( 

526 Output("risk-insight-card", "children"), 

527 [ 

528 Input("dashboard-tabs", "active_tab"), 

529 Input("region-selector", "value"), 

530 Input("demand-store", "data"),

532 ],

538 return no_update

540 

541 @app.callback( 

542 [ 

543 Output("tab5-alerts-list", "children"), 

544 Output("tab5-stress-score", "children"), 

545 Output("tab5-stress-label", "children"), 

546 Output("tab5-stress-breakdown", "children"),

560 def update_alerts_tab(region, demand_json, weather_json, active_tab): 

561 """Update Tab 5 alerts and stress indicators.""" 

562 if active_tab != "tab-alerts": 

563 return [no_update] * 8 

564 empty = _empty_figure("Loading...") 

565 

566 # ── v2 Redis fast path ────────────────────────────── 

567 if region: 

568 redis_result = _alerts_tab_from_redis(region) 

569 if redis_result is not None: 

570 return redis_result

572 # ── warming gate (REQUIRE_REDIS deployments) ──────── 

573 # The scoring job owns alert-payload generation in staging/prod; 

574 # a Redis miss there means the pipeline is warming, and the demo 

575 # fallback below must never render as real data. P2-35 (#273): 

576 # when the pipeline is demonstrably alive (fresh actuals for this 

577 # region) yet the alert payload never lands, "will populate after 

578 # the next scoring run" is a forever-lie — escalate to an honest 

579 # persistent-unavailable state instead. 

580 if REQUIRE_REDIS: 

581 from components._callbacks_shared import ( 

582 _pipeline_alive, 

583 _scoring_pass_completed_since_actuals, 

584 ) 

585 from components.error_handling import warming_state 

586 

587 # Escalate only on completed-pass evidence: within one scoring 

588 # pass a region's actuals land BEFORE its alert payload, so 

589 # fresh actuals alone would render the permanence claim during 

590 # the first pass after a flush (verification catch). 

591 if region and _pipeline_alive(region) and _scoring_pass_completed_since_actuals(region): 

592 log.info("alerts_unavailable_gate", region=region) 

593 state_card = warming_state( 

594 title="Risk data unavailable", 

595 message=( 

596 "The scoring pipeline is live, but no alert/anomaly " 

597 "payload exists for this region — its alert phase is " 

598 "not producing data. This won't resolve on its own." 

599 ), 

600 ) 

601 chip_text = "Unavailable" 

602 else: 

603 log.info("alerts_warming_gate", region=region) 

604 state_card = warming_state( 

605 title="Risk data is warming up", 

606 message=( 

607 "The scheduled pipeline has not published alert/anomaly " 

608 "data for this region yet. This page will populate after " 

609 "the next scoring run." 

610 ), 

611 ) 

612 chip_text = "Warming" 

613 return ( 

614 [state_card],

634 # Grid stress = demand / capacity supply tightness, not alert count 

635 # (#265). Same helper the scoring job calls, so the two paths cannot 

636 # disagree on the definition; they differ only in the demand reading 

637 # they have available. 

638 current_demand = None 

639 try: 

640 _dd = pd.read_json(io.StringIO(demand_json)) if demand_json else None 

641 if _dd is not None and not _dd.empty and "demand_mw" in _dd: 

642 _s = _dd["demand_mw"].dropna() 

643 current_demand = float(_s.iloc[-1]) if len(_s) else None 

644 except Exception: 

645 current_demand = None 

646 stress, stress_label = grid_stress(region, current_demand)

648 # Anomaly band: compute the +/-2 sigma envelope over the FULL series, 

649 # then slice to the displayed 168h window -- so the window is warm at 

650 # its start and the bands span the whole demand line rather than 

651 # beginning a day late. 

652 anomaly = {} 

653 if demand_json: 

654 demand_df = pd.read_json(io.StringIO(demand_json)) 

655 demand_df["timestamp"] = pd.to_datetime(demand_df["timestamp"]) 

656 roll_mean_full = demand_df["demand_mw"].rolling(24, min_periods=1).mean() 

657 roll_std_full = demand_df["demand_mw"].rolling(24, min_periods=2).std() 

658 recent = demand_df.tail(168) 

659 upper = (roll_mean_full + 2 * roll_std_full).tail(168) 

660 lower = (roll_mean_full - 2 * roll_std_full).tail(168) 

661 anomalies = recent[recent["demand_mw"] > upper] 

662 anomaly = { 

663 "timestamps": recent["timestamp"], 

664 "demand": recent["demand_mw"], 

665 "upper": upper, 

666 "lower": lower, 

667 "anomaly_timestamps": anomalies["timestamp"], 

668 "anomaly_values": anomalies["demand_mw"], 

669 } 

670 

671 temperature = {} 

672 if weather_json: 

673 weather_df = pd.read_json(io.StringIO(weather_json)) 

674 weather_df["timestamp"] = pd.to_datetime(weather_df["timestamp"]) 

675 recent_w = weather_df.tail(168) 

676 temperature = { 

677 "timestamps": recent_w["timestamp"], 

678 "values": recent_w["temperature_2m"], 

679 } 

680 

681 # Build weather context from latest reading 

682 weather_context = html.Div() 

683 if weather_json: 

684 try: 

685 w_df = pd.read_json(io.StringIO(weather_json)) 

686 if not w_df.empty: 

687 weather_context = _build_weather_context(w_df.iloc[-1]) 

688 except Exception: 

689 log.warning("weather_context_build_failed")

699 temperature=temperature, 

700 weather_context=weather_context,

711def _build_risk_insight( 

712 region: str | None, 

713 demand_json: str | None, 

714 weather_json: str | None, 

715) -> html.Div: 

716 """3-sentence narrative for the Risk tab — composes the same fragments 

717 the alerts/stress callback already computes, just rendered as prose.""" 

718 region = region or "FPL" 

719 

720 # Demand-anomaly stat: pct of last-24h hours outside ±3σ 

721 anomaly_clause = "Demand sits within normal bounds." 

722 try: 

723 if demand_json: 

724 ddf = pd.read_json(io.StringIO(demand_json)) 

725 ddf["timestamp"] = pd.to_datetime(ddf["timestamp"]) 

726 ddf = ddf.sort_values("timestamp") 

727 last_24 = ddf.tail(24)["demand_mw"].dropna() 

728 if len(last_24) >= 6: 

729 m = float(last_24.mean()) 

730 s = float(last_24.std()) or 1.0 

731 outliers = int(((last_24 - m).abs() > 2.5 * s).sum()) 

732 if outliers > 0: 

733 anomaly_clause = ( 

734 f"{outliers} hour{'s' if outliers > 1 else ''} of demand sat " 

735 "outside ±2.5σ in the last 24h." 

736 ) 

737 except Exception as exc: # pragma: no cover 

738 log.warning("risk_insight_anomaly_failed", region=region, error=str(exc)) 

739 

740 # Weather-severity stat: |temperature_2m − 65| ≥ 25 (peak heat / cold band) 

741 weather_clause = "Weather is in a comfortable band." 

742 try: 

743 if weather_json: 

744 wdf = pd.read_json(io.StringIO(weather_json)) 

745 wdf["timestamp"] = pd.to_datetime(wdf["timestamp"]) 

746 wdf = wdf.sort_values("timestamp") 

747 recent = wdf.tail(24) 

748 if "temperature_2m" in recent.columns and not recent["temperature_2m"].isna().all(): 

749 temps = recent["temperature_2m"].dropna() 

750 t_max = float(temps.max()) 

751 t_min = float(temps.min()) 

752 if t_max >= 90: 

753 weather_clause = f"Heat-driven demand risk is elevated (peak {t_max:.0f} °F)." 

754 elif t_min <= 30: 

755 weather_clause = f"Cold-driven demand risk is elevated (low {t_min:.0f} °F)." 

756 elif t_max >= 80: 

757 weather_clause = f"Temperatures trending warm (peak {t_max:.0f} °F)." 

758 elif t_min <= 40: 

759 weather_clause = f"Temperatures trending cool (low {t_min:.0f} °F)." 

760 except Exception as exc: # pragma: no cover 

761 log.warning("risk_insight_weather_failed", region=region, error=str(exc)) 

762 

763 body = [ 

764 anomaly_clause, 

765 " ", 

766 weather_clause, 

767 " Check the timeline above for active alerts.", 

768 ] 

769 return build_insight_card("Risk summary", body)

772def _build_weather_context(latest: pd.Series) -> html.Div: 

773 """Build a row of weather KPI mini-cards from the latest weather reading.""" 

774 

775 def _val(key: str) -> float | None: 

776 # None AND NaN both mean "no reading" — skip the card. (pd.Series coerces 

777 # a None to NaN, and archive-unstable columns like wind_speed_80m arrive 

778 # NaN, so a plain ``is not None`` check would render a "nan" card.) 

779 v = latest.get(key) 

780 return float(v) if v is not None and pd.notna(v) else None 

781 

782 temp = _val("temperature_2m") 

783 # 80m is the preferred anemometer height but is archive-unstable (#164); 

784 # fall through to 10m when it's missing/NaN, which get()'s default can't do. 

785 wind = _val("wind_speed_80m") 

786 if wind is None: 

787 wind = _val("wind_speed_10m") 

788 humidity = _val("relative_humidity_2m") 

789 cloud = _val("cloud_cover") 

790 

791 cards = [] 

792 

793 if temp is not None: 

794 t = float(temp) 

795 color = "#FF5C7A" if t >= 95 else ("#FFB84D" if t >= 85 else "#2BD67B") 

796 cards.append( 

797 dbc.Col( 

798 html.Div( 

799 [ 

800 html.P("TEMPERATURE", className="kpi-label"), 

801 html.H4( 

802 f"{t:.0f}°F", 

803 className="kpi-value", 

804 style={"fontSize": "1.3rem"}, 

805 ), 

806 ], 

807 className="kpi-card", 

808 style={"borderTop": f"3px solid {color}"}, 

809 ), 

810 md=3, 

811 ) 

812 ) 

813 

814 if wind is not None: 

815 w = float(wind) 

816 color = "#FF5C7A" if w >= 40 else ("#FFB84D" if w >= 25 else "#2BD67B") 

817 cards.append( 

818 dbc.Col( 

819 html.Div( 

820 [ 

821 html.P("WIND SPEED", className="kpi-label"), 

822 html.H4( 

823 f"{w:.0f} mph", 

824 className="kpi-value", 

825 style={"fontSize": "1.3rem"}, 

826 ), 

827 ], 

828 className="kpi-card", 

829 style={"borderTop": f"3px solid {color}"}, 

830 ), 

831 md=3, 

832 ) 

833 ) 

834 

835 if humidity is not None: 

836 h = float(humidity) 

837 color = "#FFB84D" if h >= 80 else "#2BD67B" 

838 cards.append( 

839 dbc.Col( 

840 html.Div( 

841 [ 

842 html.P("HUMIDITY", className="kpi-label"), 

843 html.H4( 

844 f"{h:.0f}%", 

845 className="kpi-value", 

846 style={"fontSize": "1.3rem"}, 

847 ), 

848 ], 

849 className="kpi-card", 

850 style={"borderTop": f"3px solid {color}"}, 

851 ), 

852 md=3, 

853 ) 

854 ) 

855 

856 if cloud is not None: 

857 c = float(cloud) 

858 color = "#A8B3C7" 

859 cards.append( 

860 dbc.Col( 

861 html.Div( 

862 [ 

863 html.P("CLOUD COVER", className="kpi-label"), 

864 html.H4( 

865 f"{c:.0f}%", 

866 className="kpi-value", 

867 style={"fontSize": "1.3rem"}, 

868 ), 

869 ], 

870 className="kpi-card", 

871 style={"borderTop": f"3px solid {color}"}, 

872 ), 

873 md=3, 

874 ) 

875 ) 

876 

877 if not cards: 

878 return html.Div() 

879 

880 return dbc.Row(cards, className="g-2")
