1"""Aggregate per-point weather frames into one representative frame (ADR-012). 

2 

3A BA's demand responds to weather across its whole footprint, but the 

4fetch historically sampled ONE point. The multi-point study 

5(``docs/MULTIPOINT_WEATHER_STUDY.md``) measured aggregating ~12 footprint 

6points at **+1.14 sMAPE pts** — and measured population *weighting* as 

7adding nothing (C−B ≈ 0), so this is a plain, unweighted aggregation. 

8 

9Kept out of ``data/weather_client.py`` deliberately: the per-variable 

10special cases below are the subtle part and deserve isolated tests, while 

11the client stays focused on fetch/fallback safety. 

12 

13Three rules over ``config.WEATHER_VARIABLES``: 

14 

15* ``wind_direction_10m`` — **circular mean**. An arithmetic mean of 350° 

16 and 10° is 180°, i.e. due south when the true answer is due north. 

17* ``weather_code`` — **mode**. WMO codes are ordinal categories 

18 (3=overcast, 61=rain, 71=snow, 95=thunderstorm); mean(3, 95) = 49 = 

19 "drizzle" is nonsense. 

20* everything else — **renormalizing nanmean** (see the null note below). 

21""" 

22 

23from __future__ import annotations 

24 

25import warnings 

26 

27import numpy as np 

28import pandas as pd 

29 

30from config import WEATHER_VARIABLES 

31 

32#: Degrees on a circle — an arithmetic mean is physically wrong. 

33CIRCULAR_VARS: frozenset[str] = frozenset({"wind_direction_10m"}) 

34#: Ordinal WMO categories — average is meaningless, take the mode. 

35MODE_VARS: frozenset[str] = frozenset({"weather_code"}) 

36

38class TimestampGridMismatchError(ValueError): 

39 """Per-point frames disagree on their hour grid — the caller must fail 

40 open to single-point rather than aggregate misaligned rows.""" 

41 

42 

43def aggregate_weather(frames: list[pd.DataFrame]) -> pd.DataFrame: 

44 """Collapse K per-point frames into one ``[timestamp] + 17 vars`` frame. 

45 

46 ``frames`` must share an identical timestamp grid (they come from one 

47 multi-point Open-Meteo call, so they do); a mismatch raises 

48 :class:`TimestampGridMismatchError` so the caller can fail open. 

49 

50 **Null policy — the one place this deliberately differs from the study 

51 script.** The study aggregated with ``nansum(w * arr)``, where a null 

52 point contributes 0 rather than being dropped: ``[10, 20, nan]`` 

53 averages to 10.0, not 15.0. That never bit offline (census centroids 

54 are over land, ERA5-only), but in production a grid-snapped cell can 

55 land over water or outside CONUS (NBM is CONUS-only), and a cell that 

56 starts nulling would drag every aggregated value toward zero — a 

57 silent coverage collapse of exactly the #161 flavor. So this uses a 

58 **renormalizing nanmean**: null points are dropped and the remainder 

59 averaged. An all-null hour yields NaN, which ``engineer_features`` 

60 imputes exactly as it does a single-point null today. 

61 """ 

62 if not frames: 

63 raise TimestampGridMismatchError("no frames to aggregate") 

64 if len(frames) == 1: 

65 return frames[0].copy() 

66 

67 base_ts = pd.to_datetime(frames[0]["timestamp"], utc=True) 

68 for f in frames[1:]: 

69 ts = pd.to_datetime(f["timestamp"], utc=True) 

70 if len(ts) != len(base_ts) or not ts.equals(base_ts): 

71 raise TimestampGridMismatchError( 

72 f"per-point timestamp grids differ ({len(ts)} vs {len(base_ts)} rows)" 

73 ) 

74 

75 out: dict[str, object] = {"timestamp": base_ts} 

76 for var in WEATHER_VARIABLES: 

77 if var not in frames[0].columns: 

78 continue 

79 # (n_hours, K) 

80 arr = np.column_stack( 

81 [pd.to_numeric(f[var], errors="coerce").to_numpy(dtype=float) for f in frames] 

82 ) 

83 # An all-null hour is an EXPECTED condition (the 3 vars ERA5 lacks 

84 # on deep history, or every cell outside a model's domain) and 

85 # yields NaN by design — numpy's "Mean of empty slice" warning 

86 # would be noise in the job logs, so it is suppressed here rather 

87 # than left to obscure real warnings. 

88 with np.errstate(invalid="ignore"), warnings.catch_warnings(): 

89 warnings.simplefilter("ignore", RuntimeWarning) 

90 if var in CIRCULAR_VARS: 

91 theta = np.deg2rad(arr) 

92 s = np.nanmean(np.sin(theta), axis=1) 

93 c = np.nanmean(np.cos(theta), axis=1) 

94 out[var] = np.rad2deg(np.arctan2(s, c)) % 360.0 

95 elif var in MODE_VARS: 

96 out[var] = _row_mode(arr) 

97 else: 

98 out[var] = np.nanmean(arr, axis=1) 

99 return pd.DataFrame(out) 

100 

101 

102def _row_mode(arr: np.ndarray) -> np.ndarray: 

103 """Most-common finite value per row; NaN when a row is all-null.""" 

104 picked = np.full(arr.shape[0], np.nan) 

105 for i in range(arr.shape[0]): 

106 row = arr[i] 

107 finite = row[np.isfinite(row)] 

108 if finite.size == 0: 

109 continue 

110 values, counts = np.unique(finite, return_counts=True) 

111 picked[i] = values[np.argmax(counts)] 

112 return picked
