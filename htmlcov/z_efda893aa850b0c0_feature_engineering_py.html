65def engineer_features(df: pd.DataFrame) -> pd.DataFrame: 

66 """ 

67 Compute all derived features from a merged demand + weather DataFrame. 

68 

69 Input must have columns: timestamp, demand_mw, and the 17 weather variables. 

70 Output adds 20+ new columns and drops rows with NaN. 

71 

72 Args: 

73 df: Merged DataFrame from preprocessing.merge_demand_weather(). 

74 

75 Returns: 

76 Feature-engineered DataFrame ready for model training. 

77 """ 

78 if df.empty: 

79 log.warning("feature_engineering_empty_input") 

80 return df 

81 

82 df = df.copy() 

83 log.info("feature_engineering_start", region=frame_region(df), input_rows=len(df)) 

84 

85 df = engineer_exogenous_features(df) 

86 df = add_autoregressive_demand_features(df) 

87 

88 # --- Interaction terms --- 

89 if "temperature_2m" in df.columns and "hour_sin" in df.columns: 

90 df["temp_x_hour"] = compute_temp_hour_interaction(df["temperature_2m"], df["hour_sin"]) 

91 

92 # --- Impute exogenous (weather) features, then drop only on the 

93 # demand-derived autoregressive warm-up (#161, 2026-05-29) --- 

94 # 

95 # Previously this did ``dropna(subset=<all feature cols>)``, which 

96 # dropped a row if ANY of the ~50 features was NaN. That made a single 

97 # sparse exogenous column able to collapse the entire row set: in the 

98 # 2026-05-29 incident Open-Meteo's /forecast endpoint degraded its 

99 # historical coverage and ``soil_temperature_0cm`` arrived non-null for 

100 # only ~100 of ~2100 rows — dragging every region below the 168-row 

101 # model threshold and taking down forecasts nationwide. 

102 # 

103 # A weather provider dropping one variable's coverage should degrade 

104 # that feature, not zero out all forecasts. So: impute the exogenous 

105 # (weather, raw + derived) columns — ffill → bfill → 0, the same 

106 # NaN-defense Prophet/SARIMAX already apply to their regressors at 

107 # predict time — and ``dropna`` only on the demand-derived 

108 # autoregressive features, whose sole legitimate NaN is the lag/rolling 

109 # warm-up prefix we genuinely want gone. 

110 feature_cols = _get_feature_columns(df) 

111 autoregressive = [c for c in AUTOREGRESSIVE_DEMAND_FEATURES if c in df.columns] 

112 impute_cols = [c for c in feature_cols if c not in autoregressive] 

113 if impute_cols: 

114 df[impute_cols] = df[impute_cols].ffill().bfill().fillna(0.0) 

115 

116 initial_rows = len(df) 

117 drop_subset = autoregressive or feature_cols # fall back if no AR cols present 

118 df = df.dropna(subset=drop_subset).reset_index(drop=True) 

119 dropped = initial_rows - len(df) 

120 

121 log.info( 

122 "feature_engineering_complete", 

123 # #537: without this, the dropped-row count below could not be attributed 

124 # to a BA — and dropped rows are exactly where the origin defects live. 

125 region=frame_region(df), 

126 output_rows=len(df), 

127 dropped_rows=dropped, 

128 feature_count=len(feature_cols), 

129 imputed_exogenous_cols=len(impute_cols), 

130 ) 

131 

132 return df

162def add_autoregressive_demand_features( 

163 df: pd.DataFrame, target_col: str = "demand_mw" 

164) -> pd.DataFrame: 

165 """Add demand-derived lag/rolling features that require historical demand. 

166 

167 All features are computed from rows STRICTLY BEFORE the current row — 

168 never including the current row's ``demand_mw`` value. This matches the 

169 inference-time behavior of ``compute_autoregressive_snapshot``, which 

170 sees only ``demand_history[:t]`` when computing features for time t. 

171 

172 History (2026-05-20, PR-D / #134-followup): the previous version of 

173 this function leaked the current row's target into ``ramp_rate`` 

174 (``demand[i] - demand[i-1]`` contains ``demand[i]``) and into the 

175 ``demand_roll_{N}h_*`` features (pandas' default trailing rolling 

176 window includes the current row). The model was trained on 

177 contaminated data and learned to over-rely on these features — 

178 ``demand_roll_24h_min`` was XGBoost's #2 feature for ERCOT per 

179 ``docs/BACKTEST_RESULTS.md``. At inference time the features were 

180 computed honestly via ``compute_autoregressive_snapshot``, creating a 

181 train/inference distribution shift that hurt live drift MAPE. 

182 

183 The fix is a single ``.shift(1)`` before each rolling/diff: ensures 

184 ``demand_roll_24h_mean[i]`` aggregates over ``demand[i-24..i-1]`` (the 

185 24 hours BEFORE row i), exactly matching the inference snapshot. 

186 """ 

187 df = df.copy() 

188 if target_col not in df.columns: 

189 return df 

190 

191 df["demand_lag_1h"] = compute_lag(df[target_col], periods=1) 

192 df["demand_lag_3h"] = compute_lag(df[target_col], periods=3) 

193 df["demand_lag_24h"] = compute_lag(df[target_col], periods=24) 

194 df["demand_lag_168h"] = compute_lag(df[target_col], periods=168) 

195 

196 # Pre-shifted series — every downstream rolling / diff reads from 

197 # this so the current row's demand is structurally invisible. Single 

198 # source of truth instead of repeating ``.shift(1)`` per feature. 

199 prior_demand = df[target_col].shift(1) 

200 

201 # ramp_rate[i] = demand[i-1] - demand[i-2] — matches snapshot's 

202 # ``lag_1 - lag_2``. Before this fix: demand[i] - demand[i-1] leaked. 

203 df["ramp_rate"] = compute_ramp_rate(prior_demand) 

204 

205 for window in [24, 72, 168]: 

206 prefix = f"demand_roll_{window}h" 

207 rolling = prior_demand.rolling(window=window, min_periods=1) 

208 df[f"{prefix}_mean"] = rolling.mean() 

209 df[f"{prefix}_std"] = rolling.std() 

210 df[f"{prefix}_min"] = rolling.min() 

211 df[f"{prefix}_max"] = rolling.max() 

212 

213 df["demand_momentum_short"] = compute_demand_momentum(df["demand_lag_1h"], df["demand_lag_3h"]) 

214 df["demand_momentum_long"] = compute_demand_momentum(df["demand_lag_1h"], df["demand_lag_24h"]) 

215 df["demand_ratio_24h"] = compute_demand_ratio(df["demand_lag_24h"], df["demand_roll_24h_mean"]) 

216 df["demand_ratio_168h"] = compute_demand_ratio( 

217 df["demand_lag_168h"], df["demand_roll_168h_mean"] 

218 ) 

219 return df

222def compute_autoregressive_snapshot(demand_history: list[float]) -> dict[str, float]: 

223 """Compute inference-time autoregressive features using only prior demand history.""" 

224 

225 def _lag(periods: int) -> float: 

226 return float(demand_history[-periods]) if len(demand_history) >= periods else np.nan 

227 

228 def _roll(window: int, op: str) -> float: 

229 if not demand_history: 

230 return np.nan 

231 arr = np.array(demand_history[-window:], dtype=float) 

232 if op == "mean": 

233 return float(np.mean(arr)) 

234 if op == "std": 

235 return float(np.std(arr, ddof=1)) if len(arr) > 1 else 0.0 

236 if op == "min": 

237 return float(np.min(arr)) 

238 return float(np.max(arr)) 

239 

240 lag_1 = _lag(1) 

241 lag_3 = _lag(3) 

242 lag_24 = _lag(24) 

243 lag_168 = _lag(168) 

244 lag_2 = _lag(2) 

245 roll_24_mean = _roll(24, "mean") 

246 roll_168_mean = _roll(168, "mean") 

247 

248 return { 

249 "demand_lag_1h": lag_1, 

250 "demand_lag_3h": lag_3, 

251 "demand_lag_24h": lag_24, 

252 "demand_lag_168h": lag_168, 

253 "ramp_rate": lag_1 - lag_2 if not np.isnan(lag_1) and not np.isnan(lag_2) else np.nan, 

254 "demand_momentum_short": lag_1 - lag_3 

255 if not np.isnan(lag_1) and not np.isnan(lag_3) 

256 else np.nan, 

257 "demand_momentum_long": lag_1 - lag_24 

258 if not np.isnan(lag_1) and not np.isnan(lag_24) 

259 else np.nan, 

260 "demand_ratio_24h": lag_24 / max(roll_24_mean, 1.0) 

261 if not np.isnan(lag_24) and not np.isnan(roll_24_mean) 

262 else np.nan, 

263 "demand_ratio_168h": lag_168 / max(roll_168_mean, 1.0) 

264 if not np.isnan(lag_168) and not np.isnan(roll_168_mean) 

265 else np.nan, 

266 "demand_roll_24h_mean": roll_24_mean, 

267 "demand_roll_24h_std": _roll(24, "std"), 

268 "demand_roll_24h_min": _roll(24, "min"), 

269 "demand_roll_24h_max": _roll(24, "max"), 

270 "demand_roll_72h_mean": _roll(72, "mean"), 

271 "demand_roll_72h_std": _roll(72, "std"), 

272 "demand_roll_72h_min": _roll(72, "min"), 

273 "demand_roll_72h_max": _roll(72, "max"), 

274 "demand_roll_168h_mean": roll_168_mean, 

275 "demand_roll_168h_std": _roll(168, "std"), 

276 "demand_roll_168h_min": _roll(168, "min"), 

277 "demand_roll_168h_max": _roll(168, "max"), 

278 }

292class HourIndexedHistory: 

293 """Demand keyed by hour, stored densely with NaN for hours we do not have. 

294 

295 This is the same shape the training path already works in — a continuous 

296 hourly grid where a gap is a NaN rather than an absent row — which is the 

297 point. ``add_autoregressive_demand_features`` shifts such a grid; this 

298 offsets into one. Sharing the *representation* is what makes the two paths 

299 mean the same thing, and it is cheaper than sharing a call: a window is a 

300 slice here, where a mapping would rebuild a list of up to 168 lookups on 

301 every step (measured 79x the positional path; this is 0.5x it). 

302 

303 Holes are load-bearing. ``lag_24h`` at an hour we never observed must be 

304 NaN, not the nearest thing 24 entries back — that silent reach is #559. 

305 """ 

306 

307 __slots__ = ("_base", "_dropped_writes", "_values") 

308 

309 def __init__(self, base: pd.Timestamp, values: np.ndarray) -> None: 

310 self._base = base 

311 self._values = values 

312 self._dropped_writes = 0 

313 

314 @property 

315 def dropped_writes(self) -> int: 

316 """How many writes fell outside the array and were discarded (#624). 

317 

318 Non-zero means the recursion's own later predictions did not land, so 

319 every rolling window that should have read them saw a hole instead. 

320 Counted rather than derived: the arithmetic relating this to the seed's 

321 tail gap is an off-by-one waiting to happen, and the whole point of the 

322 counter is to not have to trust it. 

323 """ 

324 return self._dropped_writes 

325 

326 @classmethod 

327 def build( 

328 cls, timestamps: Any, values: Any, *, extra_hours: int = 0 

329 ) -> "HourIndexedHistory | None": 

330 """From parallel timestamp/value sequences, or None if they do not line up. 

331 

332 ``extra_hours`` reserves room past the last seed hour for predictions the 

333 recursion will write back. 

334 """ 

335 try: 

336 stamps = pd.to_datetime(pd.Series(list(timestamps)), utc=True) 

337 vals = np.asarray(list(values), dtype=float) 

338 except (TypeError, ValueError): 

339 return None 

340 if len(stamps) != len(vals) or len(stamps) == 0: 

341 return None 

342 

343 base = stamps.min().floor("h") 

344 offsets = ((stamps - base).dt.total_seconds() // 3600).to_numpy(dtype=np.int64) 

345 size = int(offsets.max()) + 1 + max(0, extra_hours) 

346 arr = np.full(size, np.nan, dtype=float) 

347 # #129: a zero or NaN reading poisons the rolling windows, so it is not 

348 # history. Leaving it NaN is exactly "we do not have this hour". 

349 keep = np.isfinite(vals) & (vals > 0) 

350 arr[offsets[keep]] = vals[keep] 

351 return cls(base, arr) 

352 

353 @classmethod 

354 def from_mapping( 

355 cls, mapping: Mapping[pd.Timestamp, float], *, extra_hours: int = 0 

356 ) -> "HourIndexedHistory | None": 

357 if not mapping: 

358 return None 

359 return cls.build(list(mapping.keys()), list(mapping.values()), extra_hours=extra_hours) 

360 

361 def index_of(self, ts: pd.Timestamp) -> int: 

362 return int((pd.Timestamp(ts).tz_convert("UTC") - self._base).total_seconds() // 3600) 

363 

364 def set(self, ts: pd.Timestamp, value: float) -> None: 

365 """Record ``value`` at ``ts``, counting the write if it falls outside. 

366 

367 #624: ``build`` sizes the array from the last *present* seed hour, so a 

368 trailing gap leaves the recursion writing past the end. The bounds guard 

369 is correct — writing there would corrupt a neighbouring hour — but it 

370 used to be indistinguishable from a successful write. A dropped write is 

371 a counter, never a no-op. 

372 """ 

373 i = self.index_of(ts) 

374 if 0 <= i < len(self._values): 

375 self._values[i] = value 

376 else: 

377 self._dropped_writes += 1 

378 

379 def at(self, index: int) -> float: 

380 if index < 0 or index >= len(self._values): 

381 return np.nan 

382 return float(self._values[index]) 

383 

384 def lag(self, index: int) -> float: 

385 """Value at ``index``, imputed when we never observed that hour. 

386 

387 A point lag has to hold *something*: the recursion cannot skip a step, 

388 and the row build zero-fills whatever is left NaN. Zero is the worst 

389 available answer — ``demand_lag_24h = 0 MW`` is the #129 poison the seed 

390 filter exists to exclude, and the injection study measured it firing on 

391 13% of forecast steps (22.6% on IID), which is the likeliest reason that 

392 study came back against its own hypothesis 

393 (``docs/POSITIONAL_LAG_INJECTION_STUDY.md``). 

394 

395 So an absent hour is imputed, in two regimes chosen to match how demand 

396 actually behaves: 

397 

398 1. **Short hole — linear interpolation** between the observations either 

399 side. The bound is ``MAX_INTERPOLATION_GAP_HOURS`` (6), reused from 

400 ``data.preprocessing`` rather than invented here, and it covers the 

401 dominant case: 25 of the 31 real gap runs measured across the fleet 

402 are a single hour. 

403 2. **Long hole — same clock hour, previous days.** Interpolating across 

404 a 16-hour hole would smooth over a diurnal cycle, so instead step back 

405 in 24-hour multiples and take the first hour we have. Demand's 

406 dominant structure is diurnal; this preserves phase exactly, which a 

407 nearest-neighbour fill does not. 

408 

409 NaN only when neither regime finds anything — a frame that thin has 

410 bigger problems, and the caller still zero-fills it as before. 

411 

412 **This is a serve-only estimate and is out of distribution by 

413 construction.** Training drops any row whose lag source was NaN 

414 (``engineer_features``), so the model never saw an imputed lag. The goal 

415 is not to be right about the missing hour, it is to keep the row close 

416 to the distribution the model was fit on — which a plausible value does 

417 and zero does not. 

418 """ 

419 direct = self.at(index) 

420 if not np.isnan(direct): 

421 return direct 

422 

423 # Bounded scan: if either edge of the hole is further than the 

424 # interpolation limit, the hole is too long for regime 1 by definition, 

425 # so there is no reason to keep looking. 

426 span = ABSENT_HOUR_INTERPOLATION_MAX_H 

427 lo = hi = None 

428 for step in range(1, span + 1): 

429 if lo is None and not np.isnan(self.at(index - step)): 

430 lo = index - step 

431 if hi is None and not np.isnan(self.at(index + step)): 

432 hi = index + step 

433 if lo is not None and hi is not None: 

434 break 

435 if lo is not None and hi is not None and (hi - lo - 1) <= span: 

436 lo_v, hi_v = self.at(lo), self.at(hi) 

437 return float(lo_v + (hi_v - lo_v) * (index - lo) / (hi - lo)) 

438 

439 for day in range(1, ABSENT_HOUR_SEASONAL_LOOKBACK_DAYS + 1): 

440 seasonal = self.at(index - 24 * day) 

441 if not np.isnan(seasonal): 

442 return float(seasonal) 

443 return np.nan 

444 

445 def window(self, index: int, hours: int) -> np.ndarray: 

446 """The hours actually present in ``[index - hours, index)``.""" 

447 lo = max(0, index - hours) 

448 hi = max(lo, min(index, len(self._values))) 

449 seg = self._values[lo:hi] 

450 return seg[np.isfinite(seg)]

453def compute_temporal_autoregressive_snapshot( 

454 history: "HourIndexedHistory", now: pd.Timestamp 

455) -> dict[str, float]: 

456 """``compute_autoregressive_snapshot`` resolved by hour instead of position. 

457 

458 Same key set, same arithmetic. The only difference is what "24 back" means: 

459 this reads the hour ``now - 24h`` and returns NaN when we do not have it, 

460 where the positional version counts back 24 surviving entries and silently 

461 reaches further whenever the frame has a hole. 

462 

463 That difference is not hypothetical. ``engineer_features`` drops every row 

464 whose lag source was null, so ``featured`` carries real discontinuities — 

465 LGEE's ran 34 hours at a live origin, making ``demand_lag_168h`` read 

466 2026-08-10T01:00 for an origin of 2026-08-18T11:00. See #559 and 

467 ``docs/POSITIONAL_LAG_SEED_STUDY.md``. 

468 

469 Rolling windows cover the hours actually present in ``[now - window, now)`` 

470 — the temporal reading of the training side's ``min_periods=1``, which 

471 likewise tolerates a gap rather than reaching past the window to fill it. 

472 

473 Args: 

474 history: Hour-indexed demand. Real demand for seed hours, the model's 

475 own predictions for hours already forecast. 

476 now: The hour being predicted. Lags are resolved relative to this. 

477 

478 Returns: 

479 The same 21 keys ``compute_autoregressive_snapshot`` returns. 

480 """ 

481 idx = history.index_of(now) 

482 

483 def _lag(periods: int) -> float: 

484 # Imputed when absent — see ``HourIndexedHistory.lag``. Rolling windows 

485 # below deliberately do NOT impute: ``min_periods=1`` on the training 

486 # side skips a NaN inside the window rather than filling it, so 

487 # ``window()`` matching that is what keeps the two paths equivalent. 

488 return history.lag(idx - periods) 

489 

490 def _roll(window: int, op: str) -> float: 

491 arr = history.window(idx, window) 

492 if arr.size == 0: 

493 return np.nan 

494 if op == "mean": 

495 return float(arr.mean()) 

496 if op == "std": 

497 return float(arr.std(ddof=1)) if arr.size > 1 else 0.0 

498 if op == "min": 

499 return float(arr.min()) 

500 return float(arr.max()) 

501 

502 lag_1 = _lag(1) 

503 lag_2 = _lag(2) 

504 lag_3 = _lag(3) 

505 lag_24 = _lag(24) 

506 lag_168 = _lag(168) 

507 roll_24_mean = _roll(24, "mean") 

508 roll_168_mean = _roll(168, "mean") 

509 

510 return { 

511 "demand_lag_1h": lag_1, 

512 "demand_lag_3h": lag_3, 

513 "demand_lag_24h": lag_24, 

514 "demand_lag_168h": lag_168, 

515 "ramp_rate": lag_1 - lag_2 if not np.isnan(lag_1) and not np.isnan(lag_2) else np.nan, 

516 "demand_momentum_short": lag_1 - lag_3 

517 if not np.isnan(lag_1) and not np.isnan(lag_3) 

518 else np.nan, 

519 "demand_momentum_long": lag_1 - lag_24 

520 if not np.isnan(lag_1) and not np.isnan(lag_24) 

521 else np.nan, 

522 "demand_ratio_24h": lag_24 / max(roll_24_mean, 1.0) 

523 if not np.isnan(lag_24) and not np.isnan(roll_24_mean) 

524 else np.nan, 

525 "demand_ratio_168h": lag_168 / max(roll_168_mean, 1.0) 

526 if not np.isnan(lag_168) and not np.isnan(roll_168_mean) 

527 else np.nan, 

528 "demand_roll_24h_mean": roll_24_mean, 

529 "demand_roll_24h_std": _roll(24, "std"), 

530 "demand_roll_24h_min": _roll(24, "min"), 

531 "demand_roll_24h_max": _roll(24, "max"), 

532 "demand_roll_72h_mean": _roll(72, "mean"),

543def positional_seed_matches_hours( 

544 seed_timestamps: Any, origin: pd.Timestamp, *, span: int = 168 

545) -> bool: 

546 """Whether positional indexing of the seed lands on the hours it intends to. 

547 

548 This is the exact condition under which the positional and temporal 

549 recursions produce **byte-identical** forecasts, not a heuristic proxy for 

550 it. Use it to decide whether comparing the two arms can tell you anything. 

551 

552 The recursion appends its own predictions at contiguous hours, so a lag that 

553 reaches into the *predicted* part is always aligned. The only thing that can 

554 misalign an index is a hole in the **seed's tail**. Every lag and rolling 

555 window reaches at most ``span`` entries back, so it is enough that the last 

556 ``span`` seed entries are contiguous hours ending at ``origin - 1h``. 

557 

558 Returns ``True`` when the arms cannot differ (nothing to compare), ``False`` 

559 when a hole inside the lookback means they will. Unknown or unusable input 

560 returns ``False`` — "assume they differ" costs a redundant comparison, where 

561 the opposite would silently drop a real observation. 

562 """ 

563 try: 

564 stamps = pd.to_datetime(pd.Series(list(seed_timestamps)), utc=True) 

565 origin = pd.Timestamp(origin).tz_convert("UTC") 

566 except (TypeError, ValueError): 

567 return False 

568 if len(stamps) < span: 

569 return False 

570 tail = stamps.iloc[-span:] 

571 # Contiguous and ending where the forecast begins. Distinct sorted hours 

572 # spanning exactly ``span - 1`` hours cannot have a hole in them. 

573 return bool( 

574 tail.iloc[-1] == origin - pd.Timedelta(hours=1) 

575 and (tail.iloc[-1] - tail.iloc[0]) == pd.Timedelta(hours=span - 1) 

576 )

579def seed_divergence_reason( 

580 seed_timestamps: Any, origin: pd.Timestamp, *, span: int = 168 

581) -> tuple[str, int | None]: 

582 """*Why* the two seed arms differ, not just whether — and by how much (#624). 

583 

584 :func:`positional_seed_matches_hours` collapses two different situations 

585 into one ``False``, and they are not equally informative: 

586 

587 * ``hole_in_lookback`` — the seed reaches ``origin - 1h`` but has a hole 

588 further back. The arms differ, ``build`` sizes the array correctly, and 

589 the recursion writes every step it takes. **This is clean evidence about 

590 temporal indexing.** 

591 * ``seed_tail_short`` — the seed stops before ``origin - 1h``. The arms 

592 differ *and* the array is under-sized, so the tail of the horizon is 

593 silently discarded (#624). An observation here is partly about that bug. 

594 

595 Both arise from the same upstream cause — production seeds from the 

596 post-``dropna`` ``featured`` frame — which is why the second is not rare. 

597 Separating them lets a comparison be stratified rather than discarded. 

598 

599 Returns: 

600 ``(reason, tail_gap_hours)``. ``reason`` is one of ``identical``, 

601 ``hole_in_lookback``, ``seed_tail_short``, ``unusable``. 

602 ``tail_gap_hours`` is how far short of ``origin - 1h`` the seed stops, 

603 and is ``None`` when unknown. 

604 """ 

605 try: 

606 stamps = pd.to_datetime(pd.Series(list(seed_timestamps)), utc=True) 

607 origin = pd.Timestamp(origin).tz_convert("UTC") 

608 except (TypeError, ValueError): 

609 return "unusable", None 

610 if len(stamps) == 0: 

611 return "unusable", None 

612 

613 gap = int((origin - pd.Timedelta(hours=1) - stamps.iloc[-1]).total_seconds() // 3600) 

614 if positional_seed_matches_hours(stamps, origin, span=span): 

615 return "identical", gap 

616 if len(stamps) < span: 

617 return "unusable", gap 

618 return ("seed_tail_short", gap) if gap > 0 else ("hole_in_lookback", gap)

645def _warn_dropped_writes(histories: "list[HourIndexedHistory] | None") -> int: 

646 """Log once per recursion if the temporal history discarded any write (#624). 

647 

648 ``HourIndexedHistory.build`` reserves the recursion's room from the last 

649 *present* seed hour, so a trailing gap in the seed leaves the tail of the 

650 horizon writing past the end of the array. Those predictions are dropped and 

651 every rolling window that should have read them sees a hole instead. 

652 

653 Emitted at WARNING because it is silent corruption of the arm's own state, 

654 not a degraded input. Not fixed here — see #624; the sizing fix is a 

655 signature change, and this is the half that is correct either way. 

656 """ 

657 if not histories: 

658 return 0 

659 dropped = sum(h.dropped_writes for h in histories if h is not None) 

660 if dropped: 

661 log.warning( 

662 "temporal_seed_writes_dropped", 

663 dropped_writes=dropped, 

664 histories=len(histories), 

665 ) 

666 return dropped

669def recursive_autoregressive_forecast( 

670 model: Any, 

671 seed_demand: list[float] | np.ndarray | pd.Series, 

672 future_df: pd.DataFrame, 

673 predict_fn: Any, 

674 seed_timestamps: Any = None, 

675 force_temporal: bool | None = None, 

676) -> np.ndarray: 

677 """Multi-step forecast that chains its own predictions as autoregressive lags. 

678 

679 This is the honest inference protocol: each step recomputes the 

680 autoregressive features (``compute_autoregressive_snapshot``) from the 

681 growing history of *predictions* (seeded by real demand), never from 

682 observed in-window actuals. It is the single source of truth for both 

683 production scoring (``jobs.phases``) and holdout evaluation 

684 (``jobs.training_job`` / ``models.training``) so that XGBoost's reported 

685 accuracy is commensurable with Prophet/SARIMAX's multi-step holdouts and 

686 matches what production actually serves (#194/#195; #186 parity lever). 

687 

688 Args: 

689 model: A trained model accepted by ``predict_fn``. 

690 seed_demand: Real demand history strictly before the forecast window. 

691 Zero/NaN readings are filtered (a single 0 would poison the 

692 rolling-window lags — see #129). 

693 future_df: One row per forecast step, in order, carrying the non- 

694 autoregressive features (weather, calendar). Its length sets the 

695 horizon. 

696 predict_fn: ``predict_fn(model, single_row_df) -> array-like`` returning 

697 one prediction for the row. 

698 

699 Returns: 

700 1D array of length ``len(future_df)``. 

701 """ 

702 history: list[float] = [ 

703 float(v) for v in seed_demand if v is not None and not pd.isna(v) and v > 0 

704 ] 

705 preds: list[float] = [] 

706 

707 # #559: resolve lags by hour rather than by position in `history`. Off by 

708 # default and fail-open — without seed timestamps, a future frame that 

709 # carries no `timestamp`, or a length mismatch, the positional path below 

710 # runs unchanged and byte-identical. 

711 from config import feature_enabled 

712 

713 # ``force_temporal`` lets a caller pin the arm regardless of the flag — the 

714 # #559 shadow needs the treatment arm while production still serves the 

715 # control. None means "ask the flag", which is every production path. 

716 use_temporal = feature_enabled("temporal_ar_seed") if force_temporal is None else force_temporal 

717 

718 temporal_history: HourIndexedHistory | None = None 

719 step_stamps: list[pd.Timestamp] | None = None 

720 if use_temporal: 

721 step_stamps = _future_step_timestamps(future_df) 

722 temporal_history = _temporal_seed_history( 

723 seed_demand, seed_timestamps, extra_hours=len(future_df) + 1 

724 ) 

725 if temporal_history is None or step_stamps is None: 

726 temporal_history, step_stamps = None, None 

727 

728 # Per-step row construction, tuned 2026-08-05. `predict_xgboost` is 49.4% 

729 # of the scoring job's forecast phase (888.8s of 1,636.3s, 

730 # `scoring_phase_rollup`), and ~41% of THIS loop is pandas, not the model: 

731 # 763 ms/BA serial splits ~317 ms row-building / ~328 ms predict, and of 

732 # that 317 ms roughly 279 ms was the per-column `row[col] = val` loop — 

733 # ~21 separate DataFrame __setitem__ calls per step, each one re-touching 

734 # the block manager. Two changes, no contract change: 

735 # 

736 # 1. One positional `.iloc` assignment instead of ~21 __setitem__ calls. 

737 # Column positions are resolved once, on the first step; the snapshot's 

738 # KEY SET is constant across steps (only its values move), so this is 

739 # safe. `compute_autoregressive_snapshot` returns floats/NaN only, so 

740 # assigning into the float feature columns cannot change a dtype. 

741 # 2. `.ffill().bfill()` dropped. On a ONE-ROW frame both are no-ops -- 

742 # they fill along axis 0 and there is no neighbouring row to fill 

743 # from -- so the whole chain only ever did the `.fillna(0)`. This is 

744 # asserted directly in tests rather than taken on trust. 

745 # 

746 # This function is the documented single source of truth for multi-step 

747 # inference, shared by production scoring, the ADR-010 serve-path gate and 

748 # holdout evaluation (#195/#186). Its observable behaviour -- the exact 

749 # frame handed to `predict_fn` on every step -- must not move. A 

750 # differential test captures every row from both implementations and 

751 # asserts byte-equality, because "the forecasts still look right" is not 

752 # evidence at this seam. 

753 # Resolve the autoregressive columns once. Only the snapshot's VALUES move 

754 # between steps; its key set is fixed, so one probe call settles it. 

755 cols = list(future_df.columns) 

756 if temporal_history is not None and step_stamps is not None: 

757 _probe = compute_temporal_autoregressive_snapshot(temporal_history, step_stamps[0]) 

758 else: 

759 _probe = compute_autoregressive_snapshot(history) 

760 ar_keys = [k for k in _probe if k in cols] 

761 ar_positions: list[int] = [] 

762 if ar_keys: 

763 # THE DTYPE TRAP, found by the differential test and not before it: 

764 # `row[col] = <float>` REPLACED the column and implicitly upcast an int 

765 # one to float64. Positional `.iloc` assignment instead writes INTO the 

766 # existing block and raises 

767 # TypeError: Invalid value '902.51' for dtype 'int64' 

768 # This is reachable in production, not hypothetical: the tail of 

769 # `_build_future_feature_frame` does `future_df[col] = 0` for any 

770 # feature column it could not fill, which creates an int64 column. Cast 

771 # up front so the frame `predict_fn` receives is byte-identical to what 

772 # the old implementation produced — which was float64 either way. 

773 to_cast = {k: "float64" for k in ar_keys if future_df[k].dtype != np.float64} 

774 if to_cast: 

775 future_df = future_df.astype(to_cast) 

776 ar_positions = [cols.index(k) for k in ar_keys] 

777 

778 for i in range(len(future_df)): 

779 row = future_df.iloc[[i]].copy() 

780 if temporal_history is not None and step_stamps is not None: 

781 snapshot = compute_temporal_autoregressive_snapshot(temporal_history, step_stamps[i]) 

782 else: 

783 snapshot = compute_autoregressive_snapshot(history) 

784 if ar_positions: 

785 row.iloc[0, ar_positions] = [snapshot[k] for k in ar_keys] 

786 row = row.fillna(0) 

787 pred = float(predict_fn(model, row)[0]) 

788 preds.append(pred) 

789 history.append(pred) 

790 if temporal_history is not None and step_stamps is not None: 

791 temporal_history.set(step_stamps[i], pred) 

792 _warn_dropped_writes([temporal_history] if temporal_history is not None else None) 

793 return np.asarray(preds, dtype=float)

796def batched_recursive_autoregressive_forecast( 

797 model: Any, 

798 seed_demand: list[float] | np.ndarray | pd.Series, 

799 future_frames: list[pd.DataFrame], 

800 predict_fn: Any, 

801 seed_timestamps: Any = None, 

802) -> list[np.ndarray]: 

803 """``recursive_autoregressive_forecast`` for N frames that share a seed. 

804 

805 Same protocol, same per-frame results — but one ``predict_fn`` call per 

806 STEP instead of one per step *per frame*. The scenario grid (#127) runs 80 

807 weather variants off a single history, so the single-frame helper issued 

808 1,920 single-row predicts per region against production's 384: five times 

809 the whole job's predict count, for a side panel. Measured at 2.7x tick 

810 runtime and reverted (#462). The variants differ only in weather, and 

811 their step-i rows can travel through the model together. 

812 

813 The chaining is unchanged and stays per-frame: frame ``j`` appends its own 

814 prediction to its own history, so no scenario can see another's. Only the 

815 batching of the model call is shared. 

816 

817 Parity with the single-frame helper is not asserted by inspection — it is 

818 a differential test (``test_scenario_grid_batching.py``) that runs both and 

819 compares byte-for-byte, because this is the seam where "the forecasts still 

820 look right" is not evidence. 

821 

822 Args: 

823 model: A trained model accepted by ``predict_fn``. 

824 seed_demand: Real demand history strictly before the window. Filtered 

825 identically to the single-frame helper (#129). 

826 future_frames: N frames of EQUAL length, one per scenario. 

827 predict_fn: ``predict_fn(model, frame) -> array-like``, one prediction 

828 per row. Called with an N-row frame rather than a 1-row one. 

829 

830 Returns: 

831 List of N arrays, each ``len(future_frames[0])`` long, in input order. 

832 """ 

833 if not future_frames: 

834 return [] 

835 

836 horizon = len(future_frames[0]) 

837 if any(len(f) != horizon for f in future_frames): 

838 raise ValueError("every scenario frame must have the same length") 

839 

840 seed = [float(v) for v in seed_demand if v is not None and not pd.isna(v) and v > 0] 

841 n = len(future_frames) 

842 histories: list[list[float]] = [list(seed) for _ in range(n)] 

843 preds: list[list[float]] = [[] for _ in range(n)] 

844 

845 # #559, mirroring the single-frame helper. Each scenario keeps its own 

846 # timestamp-keyed history for the same reason it keeps its own list: no 

847 # scenario may see another's predictions. Fail-open to positional. 

848 from config import feature_enabled 

849 

850 temporal_histories: list[HourIndexedHistory] | None = None 

851 step_stamps: list[pd.Timestamp] | None = None 

852 if feature_enabled("temporal_ar_seed"): 

853 step_stamps = _future_step_timestamps(future_frames[0]) 

854 if step_stamps is not None: 

855 temporal_histories = [ 

856 _temporal_seed_history(seed_demand, seed_timestamps, extra_hours=horizon + 1) 

857 for _ in range(n) 

858 ] 

859 if any(h is None for h in temporal_histories): 

860 temporal_histories, step_stamps = None, None 

861 else: 

862 temporal_histories = None 

863 

864 cols = list(future_frames[0].columns) 

865 if temporal_histories is not None and step_stamps is not None: 

866 _probe = compute_temporal_autoregressive_snapshot(temporal_histories[0], step_stamps[0]) 

867 else: 

868 _probe = compute_autoregressive_snapshot(seed) 

869 ar_keys = [k for k in _probe if k in cols] 

870 

871 # Stack STEP-MAJOR: all N scenarios for step 0, then step 1, and so on. A 

872 # step's rows are then CONTIGUOUS, so each iteration is a cheap positional 

873 # slice instead of a fancy-index gather. Measured: fancy-indexing made the 

874 # batched path slower than cell-at-a-time on pandas overhead alone, which 

875 # would have wiped out the point of batching. 

876 stacked = pd.concat( 

877 [f.iloc[[i]] for i in range(horizon) for f in future_frames], ignore_index=True 

878 ) 

879 if ar_keys: 

880 # Same dtype trap as the single-frame helper: `_build_future_feature_frame` 

881 # can leave an int64 column, and positional assignment writes into the 

882 # existing block rather than replacing it. 

883 to_cast = {k: "float64" for k in ar_keys if stacked[k].dtype != np.float64} 

884 if to_cast: 

885 stacked = stacked.astype(to_cast) 

886 # Fill once up front rather than per step. The non-autoregressive columns 

887 # never change, and the autoregressive ones are overwritten every step from 

888 # the snapshot (whose own NaNs are filled in the same pass below). 

889 stacked = stacked.fillna(0) 

890 ar_positions = [cols.index(k) for k in ar_keys] 

891 block = np.empty((n, len(ar_keys)), dtype=float) if ar_keys else None 

892 

893 for i in range(horizon): 

894 step_rows = stacked.iloc[i * n : (i + 1) * n] 

895 if ar_positions: 

896 for j in range(n): 

897 if temporal_histories is not None and step_stamps is not None: 

898 snapshot = compute_temporal_autoregressive_snapshot( 

899 temporal_histories[j], step_stamps[i] 

900 ) 

901 else: 

902 snapshot = compute_autoregressive_snapshot(histories[j]) 

903 block[j] = [snapshot[k] for k in ar_keys] 

904 np.nan_to_num(block, copy=False, nan=0.0) 

905 step_rows = step_rows.copy() 

906 step_rows.iloc[:, ar_positions] = block 

907 else: 

908 step_rows = step_rows.copy() 

909 

910 out = np.asarray(predict_fn(model, step_rows), dtype=float) 

911 for j in range(n): 

912 value = float(out[j]) 

913 preds[j].append(value) 

914 histories[j].append(value) 

915 if temporal_histories is not None and step_stamps is not None: 

916 temporal_histories[j].set(step_stamps[i], value) 

917 

918 _warn_dropped_writes(temporal_histories) 

919 return [np.asarray(p, dtype=float) for p in preds]

929 Cooling Degree Days: max(0, temp - 65°F).

937 CDD series.

940

944#: actually operates on. 

945COOLING_ACCUM_WINDOWS_H = (24, 72)

948def _add_cooling_response_features(df: pd.DataFrame) -> None: 

949 """Cooling-response features, in place, behind ``cooling_response_features``. 

950 

951 Motivated by measurement, not intuition: `docs/ERROR_ANALYSIS.md` found the 

952 hottest temperature quintile carries a mean **34.7%** of our error against 

953 **11.9%** for the coldest, monotone in 7 of 8 BAs analysed. The existing 

954 representation of cooling load is a single linear ``cooling_degree_days`` 

955 against a fixed 65°F baseline. 

956 

957 Three things it cannot express, all built from weather variables we already 

958 fetch — so nothing here touches the fetch path: 

959 

960 * **Accumulation** — thermal mass across consecutive hot hours/days. 

961 * **Convexity** — cooling load rises faster than linearly in CDD once 

962 plant is near capacity. 

963 * **Humidity** — latent load; 95°F at 70% RH is not 95°F at 20%. 

964 

965 Flag-gated and default-off until the rolling-eval study says otherwise. 

966 """ 

967 from config import feature_enabled 

968 

969 if not feature_enabled("cooling_response_features"): 

970 return 

971 if "cooling_degree_days" not in df.columns: 

972 return 

973 

974 cdd = df["cooling_degree_days"] 

975 for window in COOLING_ACCUM_WINDOWS_H: 

976 df[f"cdd_accum_{window}h"] = compute_cdd_accumulation(cdd, window) 

977 # Convexity. Squared rather than a learned spline: XGBoost can already 

978 # split on CDD, so what it lacks is not flexibility but a term that makes 

979 # the curvature cheap to express in few splits. 

980 df["cdd_squared"] = cdd.pow(2) 

981 if "relative_humidity_2m" in df.columns: 

982 df["heat_index"] = compute_heat_index(df["temperature_2m"], df["relative_humidity_2m"]) 

983 # Latent load only bites when there is sensible load to add it to. 

984 df["cdd_x_humidity"] = cdd * (df["relative_humidity_2m"] / 100.0)

987def compute_heat_index(temperature_f: pd.Series, relative_humidity_pct: pd.Series) -> pd.Series: 

988 """NWS heat index (Rothfusz regression), °F. 

989 

990 Cooling load tracks what a building's occupants and its HVAC actually 

991 experience, which at high temperature is driven as much by humidity as by 

992 the dry-bulb reading: 95°F at 70% RH is a far larger air-conditioning load 

993 than 95°F at 20%. The model currently sees temperature and humidity as 

994 separate columns and has to learn the interaction from data. 

995 

996 Below 80°F the regression is not valid and the NWS uses a simple average 

997 form instead; that branch is applied here rather than extrapolating a 

998 polynomial fitted for hot, humid conditions into cool ones. 

999 """ 

1000 t = pd.to_numeric(temperature_f, errors="coerce") 

1001 r = pd.to_numeric(relative_humidity_pct, errors="coerce").clip(0, 100) 

1002 

1003 simple = 0.5 * (t + 61.0 + (t - 68.0) * 1.2 + r * 0.094) 

1004 full = ( 

1005 -42.379 

1006 + 2.04901523 * t 

1007 + 10.14333127 * r 

1008 - 0.22475541 * t * r 

1009 - 0.00683783 * t * t 

1010 - 0.05481717 * r * r 

1011 + 0.00122874 * t * t * r 

1012 + 0.00085282 * t * r * r 

1013 - 0.00000199 * t * t * r * r 

1014 ) 

1015 return simple.where(t < 80.0, full)

1018def compute_cdd_accumulation(cdd: pd.Series, window_h: int) -> pd.Series: 

1019 """Trailing mean CDD — building thermal mass, as a feature. 

1020 

1021 The third consecutive 95°F day draws materially more cooling load than the 

1022 first: structures soak heat overnight and start the next day warmer. A 

1023 point-in-time CDD cannot express that, so the model sees two identical 

1024 hours with very different true loads. 

1025 

1026 Backward-looking only (``min_periods=1`` so early rows survive rather than 

1027 poisoning the frame with NaN). 

1028 """ 

1029 return pd.to_numeric(cdd, errors="coerce").rolling(window_h, min_periods=1).mean()

1047def compute_temperature_deviation(temperature_f: pd.Series, window: int = 720) -> pd.Series: 

1048 """ 

1049 Temperature deviation from 30-day (720-hour) rolling average. 

1050 

1051 Unusual weather = unusual demand. 

1052 

1053 Args: 

1054 temperature_f: Temperature in Fahrenheit. 

1055 window: Rolling window in hours (default: 720 = 30 days). 

1056 

1057 Returns: 

1058 Deviation series. 

1059 """

1064def compute_wind_power(wind_speed_mph: pd.Series) -> pd.Series: 

1065 """ 

1066 Simplified wind power estimate: 0.5 × ρ × A × v³. 

1067 

1068 Converts mph → m/s internally. Applies cutout speed (25 m/s ≈ 56 mph). 

1069 Above cutout, turbines shut down → power = 0. 

1070 

1071 Args: 

1072 wind_speed_mph: Wind speed in mph (from Open-Meteo). 

1073 

1074 Returns: 

1075 Normalized wind power estimate [0, 1]. 

1076 """ 

1077 # Convert mph to m/s 

1078 v_ms = wind_speed_mph * MPH_TO_MS 

1079 

1080 # Simplified power curve (normalized) 

1081 # P = 0.5 * rho * A * v^3 — we normalize by rated conditions 

1082 # Using v=12 m/s as rated speed (typical for modern turbines) 

1083 rated_speed_ms = 12.0 

1084 rated_power = 0.5 * AIR_DENSITY_KG_M3 * 1.0 * (rated_speed_ms**3) 

1085 

1086 raw_power = 0.5 * AIR_DENSITY_KG_M3 * 1.0 * (v_ms**3) 

1087 normalized = raw_power / rated_power 

1088 

1089 # Apply cut-in (3 m/s) and cutout (25 m/s) speeds 

1090 cut_in_ms = 3.0 

1091 result = np.where( 

1092 v_ms < cut_in_ms, 

1093 0.0, 

1094 np.where(v_ms > WIND_CUTOUT_SPEED_MS, 0.0, np.minimum(normalized, 1.0)), 

1095 ) 

1096 

1097 return pd.Series(result, index=wind_speed_mph.index, dtype=float)
