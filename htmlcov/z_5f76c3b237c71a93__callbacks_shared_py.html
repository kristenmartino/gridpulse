1"""Shared infrastructure for ``components/callbacks.py``. 

2 

3Step 1 of the ``callbacks.py`` decomposition tracked in issue #87. This 

4module owns the cross-cutting helpers and module-level state that every 

5tab callback uses — caches, design tokens, layout helpers, data-hash 

6computation, color palettes, basemap constants.

13## Public-import surface 

14

177def _layout(*, uirevision: str | None = None, **overrides) -> dict: 

178 """Compose a Plotly layout dict from PLOT_LAYOUT + per-call overrides. 

179 

180 ``uirevision`` is the Plotly hook that preserves user-set UI state 

181 (zoom, pan, legend toggles) across figure re-renders. Tie it to 

182 *user-meaningful* state — region, horizon, model — NOT to a raw data 

183 hash, so that auto-refreshes (5-min interval, store updates) reuse 

184 the same revision string and the chart's interaction state survives. 

185 Changing the string forces Plotly to re-initialize the view, which 

186 is the desired behavior when the user picks a new region/horizon. 

187 

188 Axis dicts merge instead of replace: ``PLOT_LAYOUT`` carries the 

189 shared axis tone (gridcolor / linecolor / tickfont) and callsites 

190 layer per-chart options (titles, tick format, showgrid toggles) on 

191 top. Without this merge, a callsite that passes ``xaxis=dict(...)`` 

192 would silently lose the shared styling. 

193 """ 

194 layout = {**PLOT_LAYOUT, **overrides} 

195 # Deep-merge xaxis / yaxis (and any yaxis2/yaxis3 etc. for dual-axis 

196 # plots): combine the PLOT_LAYOUT defaults with the per-call override, 

197 # with the override winning on conflicting keys. 

198 for axis_key, default_value in PLOT_LAYOUT.items(): 

199 if not (axis_key.startswith("xaxis") or axis_key.startswith("yaxis")): 

200 continue 

201 override_value = overrides.get(axis_key) 

202 if override_value is None: 

203 continue # only the default applies — leave PLOT_LAYOUT entry alone 

204 if isinstance(default_value, dict) and isinstance(override_value, dict): 

205 layout[axis_key] = {**default_value, **override_value} 

206 # else: scalar override (rare) — already handled by the outer 

207 # ``{**PLOT_LAYOUT, **overrides}`` above. 

208 # Plotly also accepts dual-axis configs the callsite didn't have a 

209 # default for (e.g. ``yaxis2``). Those pass through unchanged from 

210 # the override dict — they're already in ``layout`` from the 

211 # outer merge above. 

212 if uirevision is not None: 

213 layout["uirevision"] = uirevision 

214 return layout

277 def _normalize_ts(ts) -> str: 

278 """Strip timezone to produce a stable string regardless of tz-aware vs tz-naive."""

284 def _frame_sig(df, key_cols: list[str]) -> dict: 

285 frame_sig: dict[str, str | int] = { 

286 "rows": int(len(df)), 

287 "start": "", 

288 "end": "", 

289 "checksum": "", 

290 } 

291 if df.empty: 

292 return frame_sig 

293 

294 if "timestamp" in df.columns: 

295 ts_bounds = pd.to_datetime(df["timestamp"], utc=True, errors="coerce") 

296 if ts_bounds.notna().any(): 

297 frame_sig["start"] = _normalize_ts(ts_bounds.min()) 

298 frame_sig["end"] = _normalize_ts(ts_bounds.max()) 

299 

300 cols = [c for c in key_cols if c in df.columns] 

301 if not cols: 

302 return frame_sig 

303 

304 sample = df.loc[:, cols].copy() 

305 if "timestamp" in sample.columns: 

306 ts = pd.to_datetime(sample["timestamp"], utc=True, errors="coerce") 

307 sample["timestamp"] = ts.astype("int64").fillna(-1).astype("int64") 

308 sample = sample.sort_values("timestamp", kind="mergesort") 

309 for col in cols: 

310 if col != "timestamp" and pd.api.types.is_numeric_dtype(sample[col]): 

311 sample[col] = sample[col].round(6) 

312 

313 hashed = pd.util.hash_pandas_object(sample.fillna("<NA>"), index=False).to_numpy( 

314 dtype=np.uint64 

315 ) 

316 frame_sig["checksum"] = f"{int(hashed.sum(dtype=np.uint64)):016x}" 

317 return frame_sig 

318 

319 signature_payload = { 

320 "region": region, 

321 "demand": _frame_sig(demand_df, ["timestamp", "demand_mw"]), 

322 "weather": _frame_sig( 

323 weather_df, 

324 [ 

325 "timestamp", 

326 "temperature_2m", 

327 "wind_speed_80m", 

328 "shortwave_radiation", 

329 "relative_humidity_2m", 

330 ], 

331 ), 

332 } 

333 signature_json = json.dumps(signature_payload, sort_keys=True, separators=(",", ":")) 

334 return hashlib.sha256(signature_json.encode("utf-8")).hexdigest()

348def _collect_backtest_residuals( 

349 region: str, 

350 model_name: str, 

351 horizon_hours: int, 

352 horizon_resolved: bool = False, 

353) -> tuple[np.ndarray, str | None]: 

354 """Collect recent backtest residuals by model/region/horizon from cache layers. 

355 

356 Returns ``(residuals, calibration_model)``. The production backtest 

357 payload only carries XGBoost predictions (``jobs/phases.py`` writer), so 

358 a request for e.g. ``"ensemble"`` residuals may be answered with a 

359 substitute model's residuals — ``calibration_model`` names the model 

360 whose residuals were actually used so callers can label the interval 

361 honestly instead of implying it was calibrated on the displayed model 

362 (2026-07 critical-review finding P1-2). Residuals from the exact 

363 requested model always take precedence over substitutes. 

364 

365 ``horizon_resolved=True`` (the #283 Phase 3b widening estimator) EXCLUDES 

366 the horizon-agnostic training-holdout source below. The flat estimator's 

367 trade-off — "the right model's residuals beat a substitute model's 

368 right-horizon residuals" — INVERTS for a lead-resolved band: pooling the 

369 same 168h holdout into every anchor collapses the per-horizon spread into 

370 a window-size artifact (a fake fan), so there the right horizon matters 

371 more than the right model, and a per-horizon substitute (disclosed via 

372 ``calibration_model``) is the honest source. 

373 """ 

374 exact_chunks: list[np.ndarray] = [] 

375 substitute_chunks: list[np.ndarray] = [] 

376 substitute_model: str | None = None 

377 

378 # In-memory cache (most recent in-process compute path) — exact matches only 

379 for (r, h, m, _mode), (cached_result, _hash, _time) in _BACKTEST_CACHE.items(): 

380 if r != region or h != horizon_hours: 

381 continue 

382 if m != model_name and not (model_name == "ensemble" and m in ("ensemble",)): 

383 continue 

384 actual = np.asarray(cached_result.get("actual", []), dtype=float) 

385 pred = np.asarray(cached_result.get("predictions", []), dtype=float) 

386 n = min(len(actual), len(pred)) 

387 if n > 0: 

388 exact_chunks.append(actual[:n] - pred[:n]) 

389 

390 # Redis pre-computed backtests (common production path) 

391 for key in ( 

392 redis_key(f"backtest:{DEFAULT_BACKTEST_EXOG_MODE}:{region}:{horizon_hours}"), 

393 redis_key(f"backtest:{region}:{horizon_hours}"), 

394 ): 

395 cached = redis_get(key) 

396 if not isinstance(cached, dict): 

397 continue 

398 actual = np.asarray(cached.get("actual", []), dtype=float) 

399 preds_map = cached.get("predictions", {}) 

400 if isinstance(preds_map, dict): 

401 if model_name in preds_map: 

402 pred = np.asarray(preds_map.get(model_name, []), dtype=float) 

403 chunk_model = model_name 

404 elif "ensemble" in preds_map: 

405 pred = np.asarray(preds_map.get("ensemble", []), dtype=float) 

406 chunk_model = "ensemble" 

407 elif preds_map: 

408 chunk_model = next(iter(preds_map.keys())) 

409 pred = np.asarray(preds_map[chunk_model], dtype=float) 

410 else: 

411 pred = np.array([], dtype=float) 

412 chunk_model = None 

413 n = min(len(actual), len(pred)) 

414 if n > 0: 

415 if chunk_model == model_name: 

416 exact_chunks.append(actual[:n] - pred[:n]) 

417 else: 

418 substitute_chunks.append(actual[:n] - pred[:n]) 

419 substitute_model = chunk_model 

420 

421 # Per-model training-holdout residuals (``gridpulse:holdout:{region}``). 

422 # The training job persists every model's holdout forecast against the 

423 # SAME actuals, so ensemble/prophet/arima self-calibrate from their OWN 

424 # residuals instead of substituting XGBoost (#196). Treated as an 

425 # exact-model source for the FLAT estimator (horizon-agnostic: a single 

426 # pooled quantile is applied across the horizon regardless, so the right 

427 # model's residuals beat a substitute model's right-horizon residuals). 

428 # Skipped when ``horizon_resolved`` — see the docstring. 

429 if not horizon_resolved: 

430 holdout = redis_get(redis_key(f"holdout:{region}")) 

431 if isinstance(holdout, dict): 

432 h_actual = np.asarray(holdout.get("actual", []), dtype=float) 

433 h_preds = holdout.get("predictions", {}) 

434 if isinstance(h_preds, dict) and model_name in h_preds: 

435 h_pred = np.asarray(h_preds.get(model_name, []), dtype=float) 

436 n = min(len(h_actual), len(h_pred)) 

437 if n > 0: 

438 exact_chunks.append(h_actual[:n] - h_pred[:n]) 

439 

440 if exact_chunks: 

441 chunks, calibration_model = exact_chunks, model_name 

442 elif substitute_chunks: 

443 chunks, calibration_model = substitute_chunks, substitute_model 

444 else: 

445 return np.array([], dtype=float), None 

446 residuals = np.concatenate(chunks) 

447 return residuals[np.isfinite(residuals)], calibration_model

450# Anchor lead times for the lead-resolved (widening) interval — mirrors 

451# ``jobs.phases.BACKTEST_HORIZONS`` (deliberately not imported: the web tier 

452# doesn't import the jobs module). Each anchor's backtest residuals estimate 

453# "typical error of an H-hour-ahead forecast", pinned at lead H. 

454_INTERVAL_ANCHOR_HORIZONS = (24, 168, 720) 

455 

456 

457def _widening_interval_from_backtests( 

458 region: str, 

459 model_name: str, 

460 target_coverage: float = 0.80, 

461) -> dict: 

462 """Lead-time-resolved empirical interval anchors (#283 Phase 3b). 

463 

464 The flat estimator below applies one q10/q90 pair — pooled across every 

465 lead time of a single backtest horizon — uniformly over the whole chart, 

466 so it reads too wide near the origin and too narrow at the deep tail. This 

467 estimator computes the same empirical quantiles **per backtest horizon** 

468 (24h / 168h / 720h) and returns them as anchors at those lead times, so 

469 the caller can interpolate a band that *widens with lead time* — the 

470 honest shape of forecast uncertainty. 

471 

472 Two statistical choices (both surfaced by the Phase 3b verification): 

473 

474 * **Horizon-resolved residuals only** (``horizon_resolved=True``): the 

475 horizon-agnostic 168h training-holdout pool is excluded — feeding the 

476 same pool into every anchor collapses the per-horizon spread into a 

477 window-size artifact (a fake fan). A per-horizon XGBoost substitute 

478 (disclosed) beats horizon-agnostic exact-model residuals here. 

479 * **Effective-lead pinning**: a horizon-H backtest pools residuals over 

480 leads 1..H, so its quantiles measure roughly the mid-window (~H/2) 

481 error, not the lead-H error. Each anchor is therefore pinned at 

482 ``effective_lead = H/2``. Leads beyond the deepest effective lead hold 

483 flat at the deepest measured value — a KNOWN-NARROW deep-tail bias 

484 (the true lead-720 error exceeds the lead-1..720 pooled quantile); 

485 preferred over pretending the pool measured lead H exactly. 

486 

487 Returns ``{"available": False}`` unless ≥2 anchors have enough residuals 

488 (the caller then falls back to the flat estimator, then the heuristic 

489 envelope). ``calibration_model`` follows the same disclosure contract as 

490 the flat estimator: exact-model calibration is claimed only when EVERY 

491 anchor used the requested model's residuals; otherwise it names the 

492 substitute so the band label can disclose it. 

493 """ 

494 from models.evaluation import empirical_error_quantiles 

495 

496 alpha = (1.0 - target_coverage) / 2.0 

497 anchors: list[dict] = [] 

498 calib_models: set[str] = set() 

499 for h in _INTERVAL_ANCHOR_HORIZONS: 

500 residuals, calib = _collect_backtest_residuals(region, model_name, h, horizon_resolved=True) 

501 if residuals.size < max(24, h // 2): 

502 continue 

503 tail_size = int(min(residuals.size, max(h * 5, 120))) 

504 q = empirical_error_quantiles(residuals[-tail_size:], lower_q=alpha, upper_q=1.0 - alpha) 

505 anchors.append( 

506 { 

507 "horizon": int(h), 

508 "effective_lead": max(1, h // 2), 

509 "lower_error": float(q["lower_error"]), 

510 "upper_error": float(q["upper_error"]), 

511 "sample_size": int(q["sample_size"]), 

512 } 

513 ) 

514 if calib is not None: 

515 calib_models.add(calib) 

516 

517 if len(anchors) < 2: 

518 return {"available": False} 

519 

520 substitutes = sorted(m for m in calib_models if m != model_name) 

521 calibration_model = model_name if not substitutes else substitutes[0] 

522 return { 

523 "available": True, 

524 "anchors": sorted(anchors, key=lambda a: a["horizon"]), 

525 "target_coverage": float(target_coverage), 

526 "calibration_model": calibration_model, 

527 }

530def _empirical_interval_from_backtests( 

531 region: str, 

532 model_name: str, 

533 horizon_hours: int, 

534 target_coverage: float = 0.80, 

535) -> dict[str, float | int | bool]: 

536 """Estimate empirical prediction interval from recent backtest residuals. 

537 

538 ``calibration_model`` in the returned dict names the model whose 

539 residuals actually calibrated the interval — it may differ from 

540 ``model_name`` (see ``_collect_backtest_residuals``), and callers must 

541 disclose that in the band label. 

542 """ 

543 from models.evaluation import empirical_error_quantiles 

544 

545 residuals, calibration_model = _collect_backtest_residuals(region, model_name, horizon_hours) 

546 if residuals.size < max(24, horizon_hours // 2): 

547 return {"available": False} 

548 

549 tail_size = int(min(residuals.size, max(horizon_hours * 5, 120))) 

550 recent = residuals[-tail_size:] 

551 alpha = (1.0 - target_coverage) / 2.0 

552 q = empirical_error_quantiles(recent, lower_q=alpha, upper_q=1.0 - alpha) 

553 return { 

554 "available": True,

598def _latest_real_demand(values, *, offset: int = 0): 

599 """Return the most recent real demand reading from a list / Series. 

600 

601 Walks backward and returns the first value that is finite (not NaN 

602 or inf) and strictly positive. ``offset=0`` finds "now"; ``offset=24`` 

603 finds "24 hours ago" while still skipping any NaN / zero rows that 

604 fall on the chosen position — so a NaN spike doesn't silently shift 

605 a trend baseline. 

606 

607 Returns ``None`` when no usable reading exists in the window — the 

608 caller should render an ``"—"`` placeholder rather than a numeric. 

609 

610 Cross-tab helper: consumed by both the Overview hero chart and the 

611 US Grid card-grid collector. Lives in shared because both tabs 

612 depend on the same NaN-guard semantics. 

613 """ 

614 if values is None: 

615 return None 

616 try: 

617 n = len(values) 

618 except TypeError: 

619 return None 

620 if n == 0: 

621 return None 

622 skipped = 0 

623 for i in range(n - 1, -1, -1): 

624 try: 

625 v = float(values.iloc[i] if hasattr(values, "iloc") else values[i]) 

626 except (TypeError, ValueError): 

627 continue 

628 if not np.isfinite(v) or v <= 0: 

629 continue 

630 if skipped < offset: 

631 skipped += 1 

632 continue 

633 return v 

634 return None

682def _pipeline_alive(region: str, max_age_hours: float = 3.0) -> bool: 

683 """True when the scoring pipeline is demonstrably writing this region — 

684 a fresh ``actuals:{region}`` ``scored_at`` within ``max_age_hours`` 

685 (the job runs hourly; 3h = several consecutive missed ticks). 

686 

687 P2-35 (#273): used to distinguish a genuine cold/warming state (nothing 

688 written yet — "will appear shortly" is honest) from a persistently 

689 unavailable surface (the pipeline IS running but this region's 

690 forecast/alert payload never lands — "shortly" was a forever-lie). 

691 Fails closed: any read/parse problem returns False, keeping the softer 

692 warming copy. 

693 """ 

694 import pandas as pd 

695 

696 from data.redis_client import redis_get, redis_key 

697 

698 try: 

699 payload = redis_get(redis_key(f"actuals:{region}")) 

700 if not isinstance(payload, dict): 

701 return False 

702 scored_at = pd.Timestamp(payload.get("scored_at")) 

703 if scored_at.tzinfo is None: 

704 scored_at = scored_at.tz_localize("UTC") 

705 age_hours = (pd.Timestamp.now(tz="UTC") - scored_at) / pd.Timedelta(hours=1) 

706 return 0 <= age_hours <= max_age_hours 

707 except Exception: 

708 return False

711def _scoring_pass_completed_since_actuals(region: str) -> bool: 

712 """True when a COMPLETE scoring pass finished at-or-after this region's 

713 current ``actuals:{region}`` write (P2-35/#273 escalation evidence). 

714 

715 The scoring job writes a region's actuals minutes BEFORE its forecast/ 

716 alert payloads within the same pass, and ``meta:last_scored`` only after 

717 ALL regions finish. Escalating to "persistently unavailable" on fresh 

718 actuals alone therefore lies during the first pass after a Redis flush 

719 (the keys land minutes later). Requiring a full pass to have completed 

720 at-or-after the actuals write proves the pipeline had its chance to 

721 produce the missing key and didn't. 

722 

723 Trade-off (documented, accepted): on a persistently-failing region the 

724 hourly refresh briefly flips this False mid-pass (~minutes, between the 

725 region's actuals write and pass end), softening the copy back to 

726 "warming" before it re-escalates — a transiently softer message, never 

727 a false permanence claim. Fails closed (False → warming) on any read 

728 or parse problem. 

729 """ 

730 import pandas as pd 

731 

732 from data.redis_client import redis_get, redis_key 

733 

734 try: 

735 meta = redis_get(redis_key("meta:last_scored")) 

736 actuals = redis_get(redis_key(f"actuals:{region}")) 

737 if not isinstance(meta, dict) or not isinstance(actuals, dict): 

738 return False 

739 completed = pd.Timestamp(meta.get("updated_at")) 

740 written = pd.Timestamp(actuals.get("scored_at")) 

741 if completed.tzinfo is None: 

742 completed = completed.tz_localize("UTC") 

743 if written.tzinfo is None: 

744 written = written.tz_localize("UTC") 

745 return bool(completed >= written) 

746 except Exception: 

747 return False
