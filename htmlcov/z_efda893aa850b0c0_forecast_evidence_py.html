8def utc_timestamp(value: object) -> str | None: 

9 """Unknown/naive clocks stay unknown; never silently assign a timezone.""" 

10 if not isinstance(value, str): 

11 return None 

12 try: 

13 parsed = datetime.fromisoformat(value.replace("Z", "+00:00")) 

14 if parsed.tzinfo is None: 

15 return None 

16 return parsed.astimezone(UTC).isoformat() 

17 except ValueError: 

18 return None

21def forecast_timing(payload: dict | None, hours: int) -> dict:

24 rows = (payload.get("forecasts") or [])[:hours]
