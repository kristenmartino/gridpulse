103def _resolve_forecast_mape(region: str) -> tuple[float | None, str]: 

104 """Resolve the most-honest MAPE to display alongside the forecast. 

105 

106 Returns ``(mape_value, source_label)`` where source_label carries both 

107 the window and the METRIC NAME actually used (e.g. ``"live 7d sMAPE"``, 

108 ``"live 30d MAPE"``, ``"holdout MAPE"``) — callers render it verbatim so 

109 an sMAPE value is never presented as MAPE: 

110 

111 - ``"live 7d …"`` — rolling 7-day live drift error from forecast-vs-actual 

112 observations stored in ``gridpulse:drift:{region}`` (the headline 

113 number; reflects how the model is actually performing right now). 

114 Prefers sMAPE (bounded, robust to near-zero-actual artifacts — #142/ 

115 PR-G9), falling back to rolling MAPE for pre-G9 payloads. 

116 - ``"live 30d"`` — rolling 30-day live drift error (fallback when 7d 

117 window has <24 records, e.g. first week post-deploy) 

118 - ``"holdout"`` — training-time holdout MAPE from each pickle's 

119 ``meta.extra["holdout_metrics"]`` (clearly labeled — this is what 

120 the model claimed at training time, not how it's doing live) 

121 - ``""`` (with value ``None``) — nothing reliable available; the 

122 forecast clause drops the MAPE annotation entirely 

123 

124 The live drift data was written hourly to Redis by the scoring job 

125 since PR #126. The Overview clause didn't read from it until this 

126 PR — it was citing training holdout MAPE instead, which was 

127 technically truthful but misleading because users read the MAPE 

128 figure as "expected accuracy of this specific forecast." 

129 """ 

130 # Layer 1: live rolling drift error. Prefer sMAPE (bounded; a near-zero 

131 # actual can't pin the headline at ~200% the way raw MAPE did for LDWP — 

132 # #142/PR-G9), falling back to the now-filtered rolling MAPE for payloads 

133 # written before G9 (which carry no sMAPE field). 

134 try: 

135 drift_payload = redis_get(redis_key(f"drift:{region}")) 

136 if isinstance(drift_payload, dict): 

137 models = drift_payload.get("models") or {} 

138 ens = models.get("ensemble") or {} 

139 # Require a meaningful window — 24 records minimum inside the 

140 # WINDOW before its figure is statistically defensible. P2-21 

141 # (#273): ``n_records`` is total retained history (trimmed by 

142 # count, not age), so gating on it let a "live 7d" headline rest 

143 # on a handful of in-window observations. Gate each window on 

144 # its own post-filter count (``n_7d``/``n_30d``); payloads 

145 # written before #273 lack the counts — fall back to the old 

146 # total-count gate for the one tick until they rewrite. 

147 n_records = int(ens.get("n_records", 0) or 0) 

148 n_7d = ens.get("n_7d") 

149 n_30d = ens.get("n_30d") 

150 ok_7d = (int(n_7d) >= 24) if n_7d is not None else n_records >= 24 

151 ok_30d = (int(n_30d) >= 24) if n_30d is not None else n_records >= 24 

152 # Track WHICH metric supplied the value — an sMAPE number must 

153 # never be labeled "MAPE" (2026-07 critical-review finding P1-8; 

154 # for artifact-prone BAs the two can differ by an order of 

155 # magnitude, e.g. LDWP sMAPE ~13% vs raw MAPE ~190%). 

156 live_7d = ens.get("rolling_smape_7d") 

157 metric_7d = "sMAPE" 

158 if live_7d is None: 

159 live_7d = ens.get("rolling_mape_7d") 

160 metric_7d = "MAPE" 

161 live_30d = ens.get("rolling_smape_30d") 

162 metric_30d = "sMAPE" 

163 if live_30d is None: 

164 live_30d = ens.get("rolling_mape_30d") 

165 metric_30d = "MAPE" 

166 if live_7d is not None and ok_7d and np.isfinite(float(live_7d)): 

167 return float(live_7d), f"live 7d {metric_7d}" 

168 if live_30d is not None and ok_30d and np.isfinite(float(live_30d)): 

169 return float(live_30d), f"live 30d {metric_30d}" 

170 except Exception as exc: # pragma: no cover — defensive 

171 log.debug("forecast_mape_drift_read_failed", region=region, error=str(exc)) 

172 

173 # Layer 2: training-time holdout MAPE (existing path), clearly labeled 

174 try: 

175 from models.model_service import get_model_metrics 

176 

177 metrics_dict = get_model_metrics(region) or {} 

178 ens_metrics = metrics_dict.get("ensemble") or metrics_dict.get("xgboost") or {} 

179 mape = ens_metrics.get("mape") 

180 if mape is not None: 

181 mape_f = float(mape) 

182 if mape_f > 0 and np.isfinite(mape_f): 

183 return mape_f, "holdout MAPE" 

184 except Exception: # pragma: no cover — defensive 

185 pass 

186 

187 return None, ""

198def _exclusions_from_freshness(freshness_data) -> list[dict]: 

199 """Extract the #309 artifact-exclusion records from the freshness store. 

200 

201 The data-freshness-store holds a JSON STRING (``json.dumps(freshness)`` in 

202 ``load_data`` — same contract ``update_fallback_banner`` parses), not a 

203 dict. The first cut of this wire checked ``isinstance(dict)`` and silently 

204 dropped the disclosure in prod while every builder-level render pin stayed 

205 green — the classic unpinned-wire gap. Tolerant of both formats and of 

206 malformed payloads; never raises. 

207 """ 

208 if not freshness_data: 

209 return [] 

210 try: 

211 parsed = json.loads(freshness_data) if isinstance(freshness_data, str) else freshness_data 

212 if isinstance(parsed, dict): 

213 exclusions = parsed.get("artifact_excluded") or [] 

214 return exclusions if isinstance(exclusions, list) else [] 

215 except (TypeError, ValueError): 

216 pass 

217 return []

220def _build_overview_metrics_items( 

221 demand_df: pd.DataFrame | None, 

222 artifact_exclusions: list[dict] | None = None, 

223) -> list[dict]: 

224 """Compose the 5-up MetricsBar cells (Now / 7d Peak / 7d Low / Average / 24h Trend).""" 

225 placeholder_labels = ["Now", "7d Peak", "7d Low", "Average", "24h Trend"] 

226 if demand_df is None or demand_df.empty or "demand_mw" not in demand_df.columns: 

227 items = [ 

228 {"label": label, "value": "—", "unit": None, "tone": "secondary"} 

229 for label in placeholder_labels 

230 ] 

231 items[0]["hero"] = True 

232 items[0]["tone"] = "primary" 

233 return items 

234 

235 df = demand_df.copy() 

236 df["timestamp"] = pd.to_datetime(df["timestamp"], utc=True) 

237 df = df.sort_values("timestamp") 

238 

239 # Strip spurious zero-demand rows (EIA's missing-observation marker) 

240 # AND NaN rows (EIA-930's publishing lag for the most recent hour, 

241 # especially for newer / smaller BAs like PSCO / NEVP / AZPS — those 

242 # rows arrive at the next hourly tick instead). The ``> 0`` check 

243 # filters both: NaN > 0 is False. 

244 nonzero = df[df["demand_mw"] > 0].reset_index(drop=True) 

245 last_7d = nonzero.tail(168) 

246 

247 # ``now_value`` reads from ``nonzero`` rather than ``df`` so the 

248 # most recent NaN / zero hour doesn't surface as "nan" / "0" in the 

249 # hero metric. Falls back to "—" when no usable reading exists. 

250 now_value = float(nonzero["demand_mw"].iloc[-1]) if not nonzero.empty else None 

251 now_ts = nonzero["timestamp"].iloc[-1] if not nonzero.empty else None 

252 peak_7d = float(last_7d["demand_mw"].max()) if not last_7d.empty else 0.0 

253 low_7d = float(last_7d["demand_mw"].min()) if not last_7d.empty else 0.0 

254 avg_7d = float(last_7d["demand_mw"].mean()) if not last_7d.empty else 0.0 

255 

256 # 24h trend (#9 — 2026-05-20) — now uses TIMESTAMP-based lookup 

257 # instead of ``iloc[-25]`` of non-zero rows. The previous index-based 

258 # approach silently compared NOW against "the 25th-from-last published 

259 # hour," which is only "24h ago" when there are zero publishing gaps 

260 # in the last 24 hours. With EIA's 1-4h publishing lag and occasional 

261 # mid-day gaps, the index approach drifted in practice. 

262 # 

263 # Trend semantics: compare NOW to the demand value at (now_ts - 24h). 

264 # Tolerance window of ±90 min absorbs single-hour publishing gaps 

265 # (EIA's most common gap profile — one missing hour, neighbors at 

266 # exactly ±60 min from the 24h-ago target). Wider gaps surface "—" 

267 # rather than fabricating a comparison against a 2h-or-more-off 

268 # anchor. 

269 trend_pct: float | None 

270 trend_anchor_ts: pd.Timestamp | None = None 

271 if now_value is not None and now_ts is not None: 

272 target_ts = now_ts - pd.Timedelta(hours=24) 

273 window_lo = target_ts - pd.Timedelta(minutes=90) 

274 window_hi = target_ts + pd.Timedelta(minutes=90) 

275 candidates = nonzero[ 

276 (nonzero["timestamp"] >= window_lo) & (nonzero["timestamp"] <= window_hi) 

277 ] 

278 if not candidates.empty: 

279 # Pick the candidate closest to the exact 24h-ago target. 

280 deltas = (candidates["timestamp"] - target_ts).abs() 

281 closest_idx = deltas.idxmin() 

282 ago_value = float(candidates.loc[closest_idx, "demand_mw"]) 

283 trend_anchor_ts = candidates.loc[closest_idx, "timestamp"] 

284 trend_pct = ((now_value - ago_value) / ago_value * 100.0) if ago_value else None 

285 else: 

286 trend_pct = None 

287 else: 

288 trend_pct = None 

289 

290 # Inverted semantic: rising demand reads as "warning" (negative tone), 

291 # falling demand reads as "positive" — matches v2 MetricsBar.tsx:64. 

292 if trend_pct is None: 

293 trend_tone = "secondary" 

294 elif trend_pct > 0.5: 

295 trend_tone = "negative" 

296 elif trend_pct < -0.5: 

297 trend_tone = "positive" 

298 else: 

299 trend_tone = "secondary" 

300 

301 now_display = f"{now_value:,.0f}" if now_value is not None else "—" 

302 trend_display = f"{trend_pct:+.1f}%" if trend_pct is not None else "—" 

303 

304 # Freshness subtext on NOW — "NOW" without context reads as 

305 # wall-clock now; in practice it's the most recent EIA-published 

306 # hour, which can be 1-4 hours behind because of EIA's publishing 

307 # lag. Make that explicit. 

308 now_subtext = f"as of {now_ts.strftime('%H:%M UTC')}" if now_ts is not None else None 

309 

310 # #309: the scoring job's quality guard excluded implausible trailing 

311 # readings (EIA partials — the LADWP "NOW 730 MW" class). The series here 

312 # is already cleaned, so the value above is the last REAL reading; this 

313 # discloses that newer-but-implausible readings exist. 

314 now_help = "Most recent actual demand reading (EIA-930)." 

315 exclusions = artifact_exclusions or [] 

316 if exclusions: 

317 n_exc = len(exclusions) 

318 plural = "s" if n_exc != 1 else "" 

319 now_subtext = ( 

320 f"{now_subtext} · {n_exc} newer reading{plural} excluded" 

321 if now_subtext 

322 else f"{n_exc} newer reading{plural} excluded" 

323 ) 

324 newest = exclusions[-1] 

325 now_help = ( 

326 f"Most recent PLAUSIBLE demand reading (EIA-930). Excluded: " 

327 f"{newest.get('mw'):,.0f} MW at {str(newest.get('ts'))[11:16]} UTC — " 

328 f"{newest.get('reason')}. EIA typically corrects these within the hour." 

329 ) 

330 

331 # Trend anchor subtext — if the 24h-ago row was off-target by more 

332 # than ~5 minutes (publishing gap absorbed by the ±30min tolerance), 

333 # surface the actual anchor time so users can see the comparison 

334 # isn't a perfect 24h. 

335 trend_subtext = None 

336 if trend_anchor_ts is not None and now_ts is not None: 

337 exact_target = now_ts - pd.Timedelta(hours=24) 

338 if abs((trend_anchor_ts - exact_target).total_seconds()) > 300: 

339 trend_subtext = f"vs {trend_anchor_ts.strftime('%H:%M UTC')}" 

340 

341 return [ 

342 { 

343 "label": "Now", 

344 "value": now_display, 

345 "unit": "MW", 

346 "hero": True, 

347 "subtext": now_subtext, 

348 "help": now_help, 

349 }, 

350 { 

351 "label": "7d Peak", 

352 "value": f"{peak_7d:,.0f}", 

353 "unit": "MW", 

354 "tone": "secondary", 

355 "subtext": "hourly max", 

356 "help": "Highest observed (actual) demand in the last 168h (7 days).", 

357 }, 

358 { 

359 "label": "7d Low", 

360 "value": f"{low_7d:,.0f}", 

361 "unit": "MW", 

362 "tone": "secondary", 

363 "subtext": "hourly min", 

364 "help": "Lowest observed (actual) demand in the last 168h (7 days).", 

365 }, 

366 { 

367 "label": "Average", 

368 "value": f"{avg_7d:,.0f}", 

369 "unit": "MW", 

370 "tone": "secondary", 

371 "subtext": "7d hourly mean", 

372 "help": "Mean observed (actual) demand over the last 168h (7 days).", 

373 }, 

374 { 

375 "label": "24h Trend", 

376 "value": trend_display, 

377 "unit": None, 

378 "tone": trend_tone, 

379 "subtext": trend_subtext, 

380 "help": "Percent change from ~24h ago to now.", 

381 }, 

382 ]

385def _build_overview_hero_chart( 

386 region: str, 

387 demand_df: pd.DataFrame | None, 

388) -> go.Figure: 

389 """7d actual demand + 24h forecast bridge with confidence band. 

390 

391 Mirrors gridpulse-v2 components/DemandChart.tsx — blue solid actual with 

392 a faint area fill, orange dashed forecast bridged from the last actual 

393 point, and an orange-tinted confidence ribbon under the forecast. 

394 """ 

395 if demand_df is None or demand_df.empty or "demand_mw" not in demand_df.columns: 

396 return _empty_figure("No demand data") 

397 

398 df = demand_df.copy() 

399 df["timestamp"] = pd.to_datetime(df["timestamp"]) 

400 df = df.sort_values("timestamp") 

401 actual = df.tail(168) 

402 if actual.empty: 

403 return _empty_figure("No recent demand") 

404 

405 # ``last_mw`` is the bridge point between the actual line and the 

406 # forecast trace. A NaN tail (EIA-930 publishing lag) would render 

407 # the bridge as a gap; instead we walk back to the most recent real 

408 # reading. ``last_ts`` matches that row so the bridge stays time-aligned. 

409 real_actual = actual[actual["demand_mw"].notna() & (actual["demand_mw"] > 0)] 

410 if real_actual.empty: 

411 return _empty_figure("No recent demand") 

412 last_ts = real_actual["timestamp"].iloc[-1] 

413 last_mw = float(real_actual["demand_mw"].iloc[-1]) 

414 

415 fig = go.Figure() 

416 

417 # Actual demand: blue solid + faint area fill below 

418 fig.add_trace( 

419 go.Scatter( 

420 x=actual["timestamp"], 

421 y=actual["demand_mw"].where(actual["demand_mw"] > 0), 

422 mode="lines", 

423 name="Actual", 

424 line=dict(color="#3b82f6", width=1.75), 

425 fill="tozeroy", 

426 fillcolor="rgba(59, 130, 246, 0.08)", 

427 hovertemplate="<b>%{x|%b %d, %H:%M}</b><br>%{y:,.0f} MW<extra></extra>", 

428 ) 

429 ) 

430 

431 # 24h forecast bridge (orange dashed) + confidence band. 

432 # 

433 # Reads the live ensemble forecast from gridpulse:forecast:{region}:1h 

434 # (written hourly by the scoring job). Falls back to actual-only chart 

435 # when Redis is cold rather than to models.model_service._simulate_forecasts, 

436 # which prior to 2026-05-20 was rendering noisy *historical* actuals at 

437 # *forward* timestamps — producing visibly wrong forecast traces (e.g. FPL 

438 # peak appearing at 04:00 instead of the real evening peak). See the 

439 # 2026-05-20 "looks off" debug + the fix branch's PR for the full 

440 # diagnosis. 

441 try: 

442 forecast_payload = _read_ensemble_forecast_from_redis(region) 

443 if forecast_payload is not None: 

444 forecast_ts_all, ensemble_arr, scored_at_iso = forecast_payload 

445 horizon = min(24, len(ensemble_arr)) 

446 forecast_ts = forecast_ts_all[:horizon] 

447 ensemble_y = list(ensemble_arr[:horizon]) 

448 

449 # Confidence band (PR-B, 2026-05-20): prefer calibrated 

450 # empirical quantiles of recent backtest residuals, fall back 

451 # to a ±3 % heuristic only when the calibration window is too 

452 # small (typically <24 residual samples — first week 

453 # post-deploy, or for newly-added regions). 

454 # 

455 # The empirical method is the same one the Forecast tab uses 

456 # (``_empirical_interval_from_backtests`` → 

457 # ``apply_empirical_interval``); see 

458 # ``components._callbacks_shared:385-409`` and 

459 # ``components._callbacks_forecast:108-170``. Sharing the 

460 # method across surfaces means both views show a band 

461 # calibrated to the same residual distribution — no 

462 # surface-specific tuning that would silently diverge. 

463 interval_meta = _empirical_interval_from_backtests(region, "ensemble", horizon) 

464 empirical_ok = bool(interval_meta.get("available")) 

465 if empirical_ok: 

466 # Additive bands: lower_error / upper_error are quantiles 

467 # of (actual - predicted) residuals, so band_y = pred + q. 

468 lower_err = float(interval_meta["lower_error"]) 

469 upper_err = float(interval_meta["upper_error"]) 

470 upper_y = [v + upper_err for v in ensemble_y] 

471 lower_y = [v + lower_err for v in ensemble_y] 

472 # Disclose the calibration source when the residuals came 

473 # from a substitute model (the prod backtest payload only 

474 # carries XGBoost predictions), so the band never implies 

475 # it was calibrated on the displayed ensemble. 

476 calib = interval_meta.get("calibration_model") 

477 calib_note = "" if calib in (None, "ensemble") else f", {calib}-calibrated" 

478 band_name = ( 

479 f"80% prediction interval " 

480 f"(empirical, n={int(interval_meta.get('sample_size', 0))}{calib_note})" 

481 ) 

482 else: 

483 upper_y = [v * 1.03 for v in ensemble_y] 

484 lower_y = [v * 0.97 for v in ensemble_y] 

485 band_name = "±3% indicative range" 

486 fig.add_trace( 

487 go.Scatter( 

488 x=list(forecast_ts) + list(forecast_ts[::-1]), 

489 y=upper_y + lower_y[::-1], 

490 fill="toself", 

491 fillcolor="rgba(249, 115, 22, 0.12)", 

492 line=dict(width=0), 

493 hoverinfo="skip", 

494 showlegend=False, 

495 name=band_name, 

496 ) 

497 ) 

498 

499 # Forecast line bridged from the last actual point — gives a 

500 # visually continuous transition. The bridge segment uses the 

501 # last actual MW; the forward segment uses the real ensemble 

502 # predictions per their Redis-stored timestamps. 

503 bridge_x = [last_ts, *forecast_ts] 

504 bridge_y = [last_mw, *ensemble_y] 

505 fig.add_trace( 

506 go.Scatter( 

507 x=bridge_x, 

508 y=bridge_y, 

509 mode="lines", 

510 name="Forecast (24h)", 

511 line=dict(color="#f97316", width=1.75, dash="dash"), 

512 hovertemplate=( 

513 "<b>%{x|%b %d, %H:%M}</b><br>%{y:,.0f} MW · forecast<extra></extra>" 

514 ), 

515 ) 

516 ) 

517 

518 # Surface the payload's own scored_at instead of discarding it — 

519 # a stale forecast must not render as the current outlook 

520 # (2026-07 critical-review finding P1-4). 

521 if scored_at_iso: 

522 scored_dt = pd.Timestamp(scored_at_iso) 

523 if scored_dt.tzinfo is None: 

524 scored_dt = scored_dt.tz_localize("UTC") 

525 age_h = (pd.Timestamp.now(tz="UTC") - scored_dt).total_seconds() / 3600.0 

526 is_stale = age_h > FRESHNESS_FRESH_MAX_AGE_HOURS 

527 note = f"forecast scored {scored_dt.strftime('%b %d %H:%M')} UTC" 

528 if is_stale: 

529 note += f" · {age_h:.0f}h ago — stale" 

530 fig.add_annotation( 

531 xref="paper", 

532 yref="paper", 

533 x=0.99, 

534 y=1.06, 

535 xanchor="right", 

536 showarrow=False, 

537 text=note, 

538 font=dict(size=10, color="#FFB84D" if is_stale else "#71717a"), 

539 ) 

540 except Exception as exc: # pragma: no cover — fall back to actual-only chart 

541 log.warning("overview_hero_forecast_failed", region=region, error=str(exc)) 

542 

543 fig.update_layout( 

544 **_layout( 

545 uirevision=region, 

546 showlegend=False, 

547 xaxis=dict( 

548 showgrid=False, 

549 linecolor="rgba(255,255,255,0.04)", 

550 tickfont=dict(color="#71717a", size=10), 

551 ), 

552 yaxis=dict( 

553 showgrid=True, 

554 gridcolor="rgba(255,255,255,0.04)", 

555 zeroline=False, 

556 tickformat=",.0f", 

557 tickfont=dict(color="#71717a", size=10), 

558 title=None, 

559 ), 

560 ), 

561 ) 

562 return fig

565def _build_overview_model_card(region: str) -> html.Div: 

566 """Horizontal model-performance bar (top/bottom borders only).""" 

567 try: 

568 from models.model_service import get_model_metrics, is_trained 

569 except ImportError: # pragma: no cover — defensive 

570 return html.Div() 

571 

572 metrics_dict = get_model_metrics(region) 

573 if not metrics_dict: 

574 return html.Div() 

575 

576 # Prefer ensemble; fall back to xgboost; finally first available 

577 if "ensemble" in metrics_dict: 

578 primary_key = "ensemble" 

579 elif "xgboost" in metrics_dict: 

580 primary_key = "xgboost" 

581 else: 

582 primary_key = next(iter(metrics_dict.keys()), None) 

583 if primary_key is None: 

584 return html.Div() 

585 

586 m = metrics_dict[primary_key] 

587 

588 def _fmt(key: str, spec: str, suffix: str = "") -> str: 

589 # An absent metric must render as unavailable, not as a perfect score. 

590 value = m.get(key) 

591 if value is None: 

592 return "—" 

593 return f"{value:{spec}}{suffix}" 

594 

595 formatted = { 

596 "MAPE": _fmt("mape", ".1f", "%"), 

597 "RMSE": _fmt("rmse", ",.0f", " MW"), 

598 "MAE": _fmt("mae", ",.0f", " MW"), 

599 "R²": _fmt("r2", ".3f"), 

600 } 

601 name = model_display_name(primary_key) 

602 badge = "trained" if is_trained(region) else "simulated" 

603 

604 caption = None 

605 if primary_key == "ensemble": 

606 single = { 

607 k: v.get("mape") 

608 for k, v in metrics_dict.items() 

609 if k != "ensemble" and v.get("mape") is not None 

610 } 

611 if single: 

612 best_key = min(single, key=single.get) 

613 best_name = model_display_name(best_key) 

614 caption = f"Production blend — most accurate single model: {best_name} {single[best_key]:.1f}%" 

615 else: 

616 caption = "Production blend — combines all models for stability." 

617 

618 return build_model_metrics_card( 

619 model_name=name, metrics=formatted, badge=badge, caption=caption 

620 )

623def _provenance_note(region: str) -> list[str]: 

624 """One class-conditional sentence from ``gridpulse:vintage_summary`` (#319). 

625 

626 Copy is deterministic and auditable: every number is a measured field from 

627 the vintage study, never generated. Silent for clean/unknown/missing. 

628 """ 

629 from components._callbacks_shared import _read_vintage_summary 

630 from config import FEED_LIMITED_MIN_REVISION_PCT 

631 

632 summary = _read_vintage_summary(region) or {} 

633 cls = summary.get("revision_class") 

634 rev = summary.get("mean_fresh_revision_pct") 

635 if not isinstance(rev, (int, float)): 

636 return [] 

637 # Same magnitude floor as the Feed-limited pills: the first live fleet 

638 # classification (2026-07-17) put half the fleet in churn/bulk at trivial 

639 # magnitudes (AVA 0.9%, AECI 0.2%, ERCOT 1.8%) — frequency-based classes 

640 # are honest, but a "revises 2%" note on thirty regions is noise, and 

641 # callouts stay credible only while they are rare. 

642 if float(rev) < FEED_LIMITED_MIN_REVISION_PCT: 

643 return [] 

644 if cls == "broken": 

645 return [ 

646 f" Data note: this region's EIA feed publishes provisional readings " 

647 f"that revise {float(rev):.0f}% on average before settling — intraday " 

648 f"figures and live accuracy firm up over the following day." 

649 ] 

650 if cls == "bulk": 

651 return [ 

652 f" Data note: same-day EIA values for this region are provisional " 

653 f"until the next-morning resubmission (measured revisions " 

654 f"{float(rev):.0f}%)." 

655 ] 

656 if cls == "churn": 

657 return [ 

658 f" Data note: EIA readings here typically revise {float(rev):.0f}% " 

659 f"shortly after publication as metering completes." 

660 ] 

661 return []

664def _build_overview_insight( 

665 region: str, 

666 demand_df: pd.DataFrame | None, 

667 persona_id: str, 

668 artifact_exclusions: list[dict] | None = None, 

669 weather_df: pd.DataFrame | None = None, 

670) -> html.Div: 

671 """Narrative paragraph with semantic-color delta spans, then role-specific lines. 

672 

673 The narrative is identical for every persona — demand level, last-24h 

674 peak, next-24h forecast with live MAPE and staleness disclosure. What 

675 varies is what follows it: ``generate_tab1_insights`` emits observations 

676 tagged with ``persona_relevance``, and ``_filter_for_persona`` keeps and 

677 ranks the ones this role cares about. Grid Ops sees ramp rate and 

678 variability; Renewables sees the temperature-demand correlation; the 

679 Data Scientist leads with variability. 

680 

681 Before #523 the persona changed only the eyebrow label — four words — 

682 which is why the landing page's "role-aware briefing" claim was reworded 

683 away in #522. This is the half that earns it back. 

684 

685 The ``peak_demand`` insight is dropped: the narrative above already 

686 states the last-24h peak, and the two together read as a stutter. Matched 

687 on ``metric_name``, not on the rendered string. 

688 """ 

689 if demand_df is None or demand_df.empty or "demand_mw" not in demand_df.columns: 

690 return build_insight_card( 

691 "Summary", 

692 ( 

693 "Awaiting demand data for this region. The forecast will populate once " 

694 "the next pipeline cycle completes." 

695 ), 

696 ) 

697 

698 df = demand_df.copy() 

699 df["timestamp"] = pd.to_datetime(df["timestamp"]) 

700 df = df.sort_values("timestamp") 

701 nonzero = df[df["demand_mw"] > 0] 

702 last_7d = nonzero.tail(168) 

703 

704 # Read from ``nonzero`` so a NaN tail (EIA-930 publishing lag) doesn't 

705 # poison the narrative with literal "nan%" deltas. 

706 now_value = _latest_real_demand(nonzero["demand_mw"]) or 0.0 

707 avg_7d = float(last_7d["demand_mw"].mean()) if not last_7d.empty else 0.0 

708 delta_pct = ((now_value - avg_7d) / avg_7d * 100.0) if avg_7d else 0.0 

709 direction = "above" if delta_pct >= 0 else "below" 

710 # Demand above/below a 7-day average is routine — neither a risk nor a win — 

711 # so render the delta in neutral text both directions (#219 — 2026-07); 

712 # alert/semantic colors are reserved for genuine risk. 

713 delta_class = "" 

714 

715 last_24h = df.tail(24) 

716 if not last_24h.empty: 

717 peak_idx = last_24h["demand_mw"].idxmax() 

718 peak_mw = float(last_24h.loc[peak_idx, "demand_mw"]) 

719 peak_ts = pd.to_datetime(last_24h.loc[peak_idx, "timestamp"]) 

720 peak_str = f"{peak_mw:,.0f} MW at {peak_ts.strftime('%H:%M')}" 

721 else: 

722 peak_str = "—" 

723 

724 # Forecast clause — read from the same Redis payload the hero chart 

725 # uses. Drop the clause entirely when Redis is cold (warming state) 

726 # instead of fabricating it from the simulated baseline. Same fix 

727 # arc as the hero chart above; see comment there for the prior-bug 

728 # context. 

729 forecast_clause = "Next-cycle forecast confidence is updating." 

730 try: 

731 forecast_payload = _read_ensemble_forecast_from_redis(region) 

732 if forecast_payload is not None: 

733 forecast_ts_all, ensemble_arr, scored_at_iso = forecast_payload 

734 horizon = min(24, len(ensemble_arr)) 

735 if horizon > 0: 

736 f_arr = ensemble_arr[:horizon] 

737 f_peak = float(f_arr.max()) 

738 f_peak_idx = int(f_arr.argmax()) 

739 # Use the REAL timestamp from the Redis payload, not a 

740 # computed offset off last_actual. The previous code 

741 # computed ``last_ts + (f_peak_idx + 1)h`` which is 

742 # meaningless against the simulated baseline (the 

743 # ensemble array represented historical hours, not 

744 # forward) and produced bogus peak times like "04:00". 

745 f_peak_ts = forecast_ts_all[f_peak_idx] 

746 

747 # MAPE for this forecast (#4 — 2026-05-20). 

748 # 

749 # Pre-fix this cited ``get_model_metrics`` which returns 

750 # training-time HOLDOUT MAPE — the model's MAPE on its 

751 # validation slice from yesterday's training run. That's 

752 # not "how this specific 24h forecast is likely to do" — 

753 # it's "how the model did on a frozen slice last night." 

754 # Users reading "MAPE 1.6%" reasonably assumed the 

755 # forecast itself was expected to be 1.6% off. Fluff. 

756 # 

757 # Honest replacement: LIVE rolling 7d MAPE from 

758 # ``gridpulse:drift:{region}`` (#121 part 1, PR #126). 

759 # This is computed by comparing every previous tick's 

760 # 1-hour-ahead forecast against the now-known actual, 

761 # rolled over the last 7 days. It tells the user how 

762 # the model has actually been performing on real 

763 # forecasts of similar horizon. 

764 # 

765 # Fall-back order: 

766 # 1. Live 7d MAPE from drift (real, calibrated to 

767 # recent reality) 

768 # 2. Live 30d MAPE from drift (more samples, slightly 

769 # staler — used only when 7d has too few records) 

770 # 3. Training holdout MAPE (clearly labeled as such) 

771 # 4. No MAPE clause at all (when nothing is available) 

772 mape_value, mape_source = _resolve_forecast_mape(region) 

773 

774 mape_clause = "" 

775 if mape_value is not None: 

776 mape_clause = f" ({mape_source} {mape_value:.1f}%)" 

777 

778 # A stale forecast must not narrate as the current outlook 

779 # (P1-4): disclose the scoring age once it exceeds the 

780 # missed-tick tolerance. 

781 stale_clause = "" 

782 if scored_at_iso: 

783 scored_dt = pd.Timestamp(scored_at_iso) 

784 if scored_dt.tzinfo is None: 

785 scored_dt = scored_dt.tz_localize("UTC") 

786 age_h = (pd.Timestamp.now(tz="UTC") - scored_dt).total_seconds() / 3600.0 

787 if age_h > FRESHNESS_FRESH_MAX_AGE_HOURS: 

788 stale_clause = f" Forecast last refreshed {age_h:.0f}h ago." 

789 

790 forecast_clause = ( 

791 f"Next-24h forecast peaks at {f_peak:,.0f} MW around " 

792 f"{f_peak_ts.strftime('%H:%M')} UTC{mape_clause}.{stale_clause}" 

793 ) 

794 except Exception as exc: # pragma: no cover 

795 log.warning("overview_insight_forecast_failed", region=region, error=str(exc)) 

796 

797 # #8 (2026-05-20): relabel "Recent peak" → "Last 24h peak" so the 

798 # window is explicit and consistent with the "7d Peak" cell in the 

799 # metrics bar above. The previous label was ambiguous about whether 

800 # "recent" meant 24h or the current 7d-peak window. 

801 body = [ 

802 "Demand is ", 

803 html.Span(f"{abs(delta_pct):.1f}% {direction}", className=delta_class), 

804 " the 7-day average. ", 

805 "Last 24h peak: ", 

806 html.Span(peak_str, className="gp-insight-card__strong"), 

807 ". ", 

808 forecast_clause, 

809 ] 

810 # #309: name the exclusion in prose — the tiles above already computed on 

811 # the cleaned series, so without this sentence the exclusion is invisible. 

812 for exc in (artifact_exclusions or [])[-1:]: 

813 body.append( 

814 f" Latest EIA reading ({exc.get('mw'):,.0f} MW at " 

815 f"{str(exc.get('ts'))[11:16]} UTC) excluded: {exc.get('reason')} — " 

816 f"an EIA reporting artifact, not demand." 

817 ) 

818 

819 # PR 3: the per-class data note, from the vintage study's measured 

820 # verdict. clean/unknown/missing → silence — callouts stay rare, which is 

821 # what keeps them credible. 

822 body.extend(_provenance_note(region)) 

823 

824 # #523: the role-aware half. Appended as sentences rather than a bullet 

825 # list because ``build_insight_card`` renders ``body`` inside an 

826 # ``html.P`` — nesting a ``<ul>`` there is invalid HTML and browsers 

827 # break the paragraph around it. 

828 try: 

829 from components.insights import generate_tab1_insights 

830 

831 for ins in generate_tab1_insights(persona_id, region, demand_df, weather_df): 

832 # The narrative already reported the peak two sentences ago. 

833 if ins.metric_name == "peak_demand": 

834 continue 

835 body.append(f" {ins.text}") 

836 except Exception as exc: # pragma: no cover — never let this drop the card 

837 log.warning("overview_insight_persona_lines_failed", region=region, error=str(exc)) 

838 

839 # Eyebrow names the role; the lines above are what make it true. 

840 eyebrow_map = { 

841 "grid_ops": "Operating summary", 

842 "renewables": "Renewables outlook", 

843 "trader": "Market signal", 

844 "data_scientist": "Model summary", 

845 } 

846 eyebrow = eyebrow_map.get(persona_id, "Summary") 

847 return build_insight_card(eyebrow, body)
