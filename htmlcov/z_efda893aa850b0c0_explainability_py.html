89def _label(feature_name: str) -> str: 

90 """Look up human label for a feature, falling back to title-cased name.""" 

91 return FEATURE_LABELS.get(feature_name, feature_name.replace("_", " ").title())

94def get_top_drivers_shap( 

95 shap_values: np.ndarray, 

96 feature_names: list[str], 

97 index: int, 

98 top_n: int = 3, 

99) -> list[tuple[str, float]]: 

100 """Return top N (label, shap_value_mw) for a single forecast point. 

101 

102 Args: 

103 shap_values: Shape (n_points, n_features). 

104 feature_names: Feature column names matching axis 1. 

105 index: Row index of the forecast point. 

106 top_n: Number of top drivers to return. 

107 

108 Returns: 

109 List of (human_label, shap_mw) sorted by absolute impact descending. 

110 """ 

111 row = shap_values[index] 

112 abs_vals = np.abs(row) 

113 top_indices = np.argsort(abs_vals)[::-1][:top_n] 

114 return [(_label(feature_names[i]), float(row[i])) for i in top_indices]

117def get_top_drivers_global( 

118 model_dict: dict, 

119 top_n: int = 3, 

120) -> list[tuple[str, float]]: 

121 """Return top N (label, importance) from XGBoost feature importances. 

122 

123 Args: 

124 model_dict: Output from train_xgboost() containing 'model' and 'feature_names'. 

125 top_n: Number of top features to return. 

126 

127 Returns: 

128 List of (human_label, importance) sorted descending. 

129 """ 

130 model = model_dict["model"] 

131 feature_names = model_dict["feature_names"] 

132 importances = model.feature_importances_ 

133 top_indices = np.argsort(importances)[::-1][:top_n] 

134 return [(_label(feature_names[i]), float(importances[i])) for i in top_indices]

137def format_driver_line(label: str, value: float, mode: str = "shap") -> str: 

138 """Format a single driver for tooltip display. 

139 

140 Args: 

141 label: Human-readable feature name. 

142 value: SHAP value in MW (mode="shap") or importance score (mode="global"). 

143 mode: "shap" for per-point MW values, "global" for importance ranking. 

144 

145 Returns: 

146 Formatted string like "Temperature: +1,200 MW" or "Temperature (high)". 

147 """ 

148 if mode == "shap": 

149 sign = "+" if value >= 0 else "" 

150 return f"{label}: {sign}{value:,.0f} MW" 

151 return f"{label} (high)" if value > 0 else label

154def build_tooltip_strings( 

155 shap_values: np.ndarray | None, 

156 feature_names: list[str] | None, 

157 model_dict: dict | None, 

158 n_points: int, 

159 model_name: str, 

160 top_n: int = 3, 

161) -> list[str]: 

162 """Build a list of tooltip strings, one per forecast point. 

163 

164 XGBoost with SHAP data → per-point drivers with MW values. 

165 Other models with model_dict → static global importance repeated. 

166 Fallback → empty strings (no extra tooltip content). 

167 

168 Args: 

169 shap_values: SHAP values array (n_points, n_features) or None. 

170 feature_names: Feature names matching SHAP columns. 

171 model_dict: XGBoost model dict for global importance fallback. 

172 n_points: Number of forecast points. 

173 model_name: Model name string. 

174 top_n: Number of top drivers per point. 

175 

176 Returns: 

177 List of n_points tooltip strings. 

178 """ 

179 empty = [""] * n_points 

180 

181 # XGBoost with per-point SHAP 

182 if ( 

183 model_name == "xgboost" 

184 and shap_values is not None 

185 and feature_names is not None 

186 and len(shap_values) >= n_points 

187 ): 

188 tooltips = [] 

189 for i in range(n_points): 

190 drivers = get_top_drivers_shap(shap_values, feature_names, i, top_n) 

191 lines = [format_driver_line(lbl, val, mode="shap") for lbl, val in drivers] 

192 tooltips.append("<br>".join(lines)) 

193 log.info("tooltips_built", mode="shap", n_points=n_points) 

194 return tooltips 

195 

196 # Global importance fallback (any model, if we have an XGBoost model_dict) 

197 if model_dict is not None: 

198 try: 

199 drivers = get_top_drivers_global(model_dict, top_n) 

200 lines = [format_driver_line(lbl, val, mode="global") for lbl, val in drivers] 

201 static_tip = "<br>".join(lines) 

202 log.info("tooltips_built", mode="global", n_points=n_points) 

203 return [static_tip] * n_points 

204 except Exception: 

205 log.debug("tooltip_global_fallback_failed") 

206 return empty 

207 

208 return empty
