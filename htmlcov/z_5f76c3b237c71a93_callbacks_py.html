136def _freshness_from_payload(payload, now) -> str | None: 

137 """Measure a payload's freshness from its own ``scored_at`` stamp. 

138 

139 Returns ``"fresh"``/``"stale"``, or ``None`` when the payload carries no 

140 (parseable) stamp — callers then apply a payload-specific fallback. 

141 Freshness must be measured, never asserted at render time (2026-07 

142 critical-review finding P1-3). 

143 """ 

144 if not isinstance(payload, dict): 

145 return None 

146 iso = payload.get("scored_at") 

147 if not iso: 

148 return None 

149 try: 

150 scored = pd.Timestamp(iso) 

151 if scored.tzinfo is None: 

152 scored = scored.tz_localize("UTC") 

153 except (ValueError, TypeError): 

154 return None 

155 age_h = (now - scored).total_seconds() / 3600.0 

156 return "fresh" if age_h <= FRESHNESS_FRESH_MAX_AGE_HOURS else "stale"

159def _load_data_from_redis(region): 

160 """Redis fast path for load_data callback. 

161 

162 Returns a 5-tuple (demand_json, weather_json, freshness_json, audit_json, 

163 pipeline_json) when Redis has both actuals and weather for the region, 

164 or None if either cache miss occurs. 

165 """ 

166 import json 

167 from datetime import UTC, datetime 

168 

169 from data.audit import audit_trail 

170 from observability import PipelineLogger 

171 

172 cached_actuals = redis_get(redis_key(f"actuals:{region}")) 

173 cached_weather = redis_get(redis_key(f"weather:{region}")) 

174 if cached_actuals is None or cached_weather is None: 

175 return None 

176 

177 pipe = PipelineLogger("load_data", region=region) 

178 now = pd.Timestamp.now(tz="UTC") 

179 # Statuses are filled in below from each payload's own age — the 

180 # defaults here are the pessimistic values for payloads we can't date. 

181 freshness = { 

182 "demand": "stale", 

183 "weather": "stale", 

184 "alerts": "warming", 

185 "timestamp": datetime.now(UTC).isoformat(), 

186 } 

187 

188 log.info("load_data_redis_hit", region=region) 

189 pipe.step("fetch_demand", rows=len(cached_actuals.get("demand_mw", [])), source="redis") 

190 pipe.step("fetch_weather", rows=len(cached_weather.get("timestamps", [])), source="redis") 

191 

192 # Convert parallel-arrays to DataFrame JSON. Sanitize spurious zero-demand 

193 # readings to NaN — a balancing authority never truly reads 0 MW; these 

194 # are missing-data artifacts (commonly legacy Redis entries written 

195 # before the EIA ingestion fix). Charts/KPIs handle NaN correctly; zeros 

196 # otherwise render as misleading dips to zero. 

197 demand_df = pd.DataFrame( 

198 { 

199 "timestamp": cached_actuals["timestamps"], 

200 "demand_mw": cached_actuals["demand_mw"], 

201 } 

202 ) 

203 demand_df.loc[demand_df["demand_mw"] <= 0, "demand_mw"] = np.nan 

204 weather_cols = {k: v for k, v in cached_weather.items() if k not in ("region", "scored_at")} 

205 weather_df = pd.DataFrame(weather_cols) 

206 if "timestamps" in weather_df.columns: 

207 weather_df = weather_df.rename(columns={"timestamps": "timestamp"}) 

208 

209 # Measured freshness (P1-3). Payloads carry scored_at since 2026-07; 

210 # legacy demand payloads fall back to the age of the newest observation 

211 # with EIA publishing-lag allowance (~1-4h is normal; see #129). 

212 demand_status = _freshness_from_payload(cached_actuals, now) 

213 if demand_status is None and len(demand_df) > 0: 

214 last_obs = pd.Timestamp(demand_df["timestamp"].iloc[-1]) 

215 if last_obs.tzinfo is None: 

216 last_obs = last_obs.tz_localize("UTC") 

217 age_h = (now - last_obs).total_seconds() / 3600.0 

218 demand_status = "fresh" if age_h <= FRESHNESS_DEMAND_LAG_ALLOWANCE_HOURS else "stale" 

219 freshness["demand"] = demand_status or "stale" 

220 freshness["weather"] = _freshness_from_payload(cached_weather, now) or freshness["demand"] 

221 # #309: readings the scoring job's quality guard excluded from the series 

222 # (already NaN in demand_mw above) — threaded to the Overview so the NOW 

223 # tile and operating summary can DISCLOSE the exclusion, not just skip it. 

224 freshness["artifact_excluded"] = cached_actuals.get("artifact_excluded") or [] 

225 

226 cached_alerts = redis_get(redis_key(f"alerts:{region}")) 

227 if isinstance(cached_alerts, dict): 

228 alerts_src = cached_alerts.get("alerts_source", "demo") 

229 if alerts_src in ("unavailable", "demo"): 

230 freshness["alerts"] = alerts_src 

231 else: 

232 freshness["alerts"] = _freshness_from_payload(cached_alerts, now) or "fresh" 

233 

234 if len(demand_df) > 0: 

235 freshness["latest_data"] = str(demand_df["timestamp"].iloc[-1]) 

236 

237 pipe.step( 

238 "serialize", 

239 demand_cols=len(demand_df.columns), 

240 weather_cols=len(weather_df.columns), 

241 ) 

242 audit_record = audit_trail.record_forecast( 

243 region=region, 

244 demand_source="redis", 

245 weather_source="redis", 

246 demand_rows=len(demand_df), 

247 weather_rows=len(weather_df), 

248 demand_range=( 

249 str(demand_df["timestamp"].iloc[0]), 

250 str(demand_df["timestamp"].iloc[-1]), 

251 ) 

252 if len(demand_df) > 0 

253 else ("", ""), 

254 weather_range=( 

255 str(weather_df["timestamp"].iloc[0]), 

256 str(weather_df["timestamp"].iloc[-1]), 

257 ) 

258 if "timestamp" in weather_df.columns and len(weather_df) > 0 

259 else ("", ""), 

260 forecast_source="redis", 

261 ) 

262 pipe.step("audit_recorded", record_id=audit_record.record_id) 

263 pipeline_summary = pipe.done() 

264 return ( 

265 demand_df.to_json(date_format="iso"), 

266 weather_df.to_json(date_format="iso"), 

267 json.dumps(freshness), 

268 audit_record.to_json(), 

269 json.dumps(pipeline_summary, default=str), 

270 )
