1""" 

2Thin Redis client for dual-mode Dash callbacks. 

3 

4When REDIS_HOST is set, reads pre-computed data from Redis (v2 pipeline). 

5When REDIS_HOST is empty or Redis is unreachable, returns None so 

6callbacks fall through to the existing v1 compute path. 

7 

8Both paths have two flavours, split along the same line (#268 / #313): 

9 

10* **Fail-soft** — ``redis_get`` returns None on any error, ``redis_set`` 

11 returns False. For the web tier, where a Redis blip should degrade to the 

12 warming state, never crash a callback. 

13* **Fail-loud** — ``redis_get_strict`` raises :class:`RedisReadError` so that 

14 *absence* and *failure* are different answers; ``persist`` raises 

15 :class:`RedisWriteError` on a dropped write. For stateful read-modify-write 

16 consumers (the scoring-job window phases), where mistaking an outage for 

17 "no history" destructively rebuilds state — the #313 vintage re-pin. 

18""" 

19 

20import json 

21import logging 

22import os 

23import threading 

24import time 

25 

26import structlog 

27 

28logger = logging.getLogger(__name__) 

29#: structlog alongside the stdlib logger: only structlog output carries 

30#: ``jsonPayload.event``, which is what log-based alert policies match on. 

31_log = structlog.get_logger() 

32 

33_redis_client = None 

34#: Monotonic time of the last *failed/absent* connection attempt. Before #268 

35#: a single failed first ping pinned the client off for the whole process 

36#: (``_redis_init_attempted`` was never reset), so a momentary blip at job start 

37#: silenced Redis for the entire scoring tick. Now a failure is retried at most 

38#: once per ``_REDIS_RETRY_INTERVAL_S`` so it self-heals mid-run. 

39_redis_last_attempt = 0.0 

40_REDIS_RETRY_INTERVAL_S = 30.0 

41#: Serializes the FIRST connect so concurrent callers wait (~1s) for the 

42#: winner instead of taking the silent backoff-None. Before #313's trigger 

43#: fix, a scoring container's 51 worker threads raced this init at cold 

44#: start: the first thread set ``_redis_last_attempt`` and spent ~1s 

45#: connecting over the VPC, and every thread arriving inside that window got 

46#: a silent ``None`` — indistinguishable from "Redis is down", with no log. 

47#: That None is what destructively re-pinned four regions' vintage windows 

48#: on 2026-07-16 (caught live by the #314 tripwire at 09:00:52.85-.88Z on 

49#: 2026-07-17: three failures 30ms apart, CAISO succeeding 1.5s later). 

50_redis_init_lock = threading.Lock()

53def _get_redis(): 

54 """Lazy-init a Redis connection. Returns client or None. 

55 

56 A healthy client is cached and reused. The first connect is serialized 

57 behind ``_redis_init_lock`` — concurrent callers block briefly for the 

58 winner's result rather than being told "no Redis" while a connect is in 

59 flight (#313). A genuinely FAILED attempt is re-probed at most once per 

60 ``_REDIS_RETRY_INTERVAL_S`` (backoff), so a transient blip recovers 

61 within the same process instead of pinning Redis off for its lifetime 

62 (#268 / P2-03). 

63 """ 

64 global _redis_client, _redis_last_attempt 

65 

66 if _redis_client is not None: 

67 return _redis_client 

68 

69 with _redis_init_lock: 

70 # Double-checked: the winner may have connected while we waited. 

71 if _redis_client is not None: 

72 return _redis_client 

73 

74 now = time.monotonic() 

75 if _redis_last_attempt and (now - _redis_last_attempt) < _REDIS_RETRY_INTERVAL_S: 

76 return None # recent FAILED attempt — back off rather than re-probe every call 

77 _redis_last_attempt = now 

78 

79 host = os.getenv("REDIS_HOST", "") 

80 if not host: 

81 logger.debug("REDIS_HOST not set — v1 compute mode") 

82 return None 

83 

84 port = int(os.getenv("REDIS_PORT", "6379")) 

85 try: 

86 import redis 

87 

88 client = redis.Redis( 

89 host=host, 

90 port=port, 

91 decode_responses=True, 

92 socket_connect_timeout=2, 

93 socket_timeout=2, 

94 ) 

95 client.ping() 

96 _redis_client = client 

97 # A SUCCESSFUL connect clears the attempt stamp so an unrelated 

98 # later re-init (tests, client reset) is not backoff-blocked. 

99 _redis_last_attempt = 0.0 

100 logger.info("Redis connected: %s:%s", host, port) 

101 return _redis_client 

102 except Exception as exc: 

103 logger.warning("Redis unavailable (%s:%s): %s — falling back to v1", host, port, exc) 

104 return None

107def redis_key(suffix: str) -> str: 

108 """Compose a fully-qualified Redis key from the configured prefix and a suffix. 

109 

110 Callers pass the part of the key *after* the prefix, e.g. 

111 ``redis_key("actuals:FPL")`` returns ``"gridpulse:actuals:FPL"`` 

112 (default). Putting the indirection in this module — rather than at 

113 every callsite — means future renames are a single-line change. 

114 

115 The prefix is read every call from ``config.REDIS_KEY_PREFIX``, so 

116 tests can override it via ``importlib.reload(config)`` after 

117 monkeypatching the env. In production it's effectively constant per 

118 process — Cloud Run revisions get their prefix from the env on first 

119 request and don't see env changes until the next deploy. 

120 

121 Issue #91 tracked the original ``wattcast`` → ``gridpulse`` rename; 

122 this helper was introduced as part of that work and stays useful as 

123 a hedge against the next rename. 

124 """ 

125 # Imported lazily to avoid a circular: config -> nothing, this module 

126 # -> config is fine, but importing at module top would force config 

127 # to load before logging is configured in tests that monkeypatch 

128 # ``os.environ``. 

129 from config import REDIS_KEY_PREFIX 

130 

131 return f"{REDIS_KEY_PREFIX}:{suffix}"

134def redis_get(key: str) -> dict | list | None: 

135 """Read a JSON value from Redis. Returns parsed object or None.""" 

136 client = _get_redis() 

137 if client is None: 

138 return None 

139 try: 

140 raw = client.get(key) 

141 if raw is None: 

142 return None 

143 return json.loads(raw) 

144 except Exception as exc: 

145 logger.warning("Redis read error for %s: %s", key, exc) 

146 return None

149class RedisReadError(RuntimeError): 

150 """A Redis read failed for infrastructure reasons — client unavailable, 

151 connection/command error, or an unparseable payload. Distinct from a key 

152 that is genuinely absent (a nil reply on a healthy connection). 

153 

154 The read-side twin of :class:`RedisWriteError` (#268). Exists because of 

155 #313: ``redis_get`` collapses failure and absence into one ``None``, and a 

156 stateful consumer that mistakes an outage for "no history" destructively 

157 rebuilds its window — prod re-pinned four regions' vintage first-sight 

158 records exactly this way on 2026-07-16. 

159 """

162def redis_get_strict(key: str) -> dict | list | None: 

163 """Read a JSON value, refusing to conflate failure with absence (#313). 

164 

165 Returns the parsed object, or ``None`` **only** when Redis affirmatively 

166 reports the key absent. Every other outcome — no client, command error, 

167 unparseable payload — raises :class:`RedisReadError`. Callers whose next 

168 action differs between "the key is not there" and "I could not find out" 

169 (window phases doing read-modify-write) must use this instead of 

170 :func:`redis_get`. 

171 """ 

172 client = _get_redis() 

173 if client is None: 

174 raise RedisReadError(f"redis unavailable for read of {key}") 

175 try: 

176 raw = client.get(key) 

177 except Exception as exc: 

178 raise RedisReadError(f"read failed for {key}: {exc}") from exc 

179 if raw is None: 

180 return None 

181 try: 

182 return json.loads(raw) 

183 except (TypeError, ValueError) as exc: 

184 # A value that exists but cannot be parsed is a failure, not an 

185 # absence — overwriting it would destroy whatever it was. 

186 raise RedisReadError(f"unparseable payload at {key}: {exc}") from exc

199#: Failed `redis_set` calls this process. Process-local and drained once per 

200#: run, like `data.eia_client`'s latency stats. 

201#: 

202#: A write failure used to be invisible three times over: `redis_set` swallowed 

203#: the exception, all 15 job call sites ignored the returned False, and the one 

204#: warning it emitted went through stdlib `logging` — so it arrived in Cloud 

205#: Logging as `textPayload` with no `jsonPayload.event`, which no log-based 

206#: alert can match. That is the same defect documented in 

207#: docs/monitoring/README.md for the job logs (inert until 2026-07-15). 

208#: A scoring run could report a region scored while writing nothing. 

209_write_failures: list[str] = [] 

210_write_failure_lock = threading.Lock() 

211 

212 

213def drain_write_failures() -> dict | None: 

214 """Return and clear this process's failed-write tally. 

215 

216 None when nothing failed, so a caller can skip the log line rather than 

217 publish an empty one. 

218 """ 

219 with _write_failure_lock: 

220 keys = list(_write_failures) 

221 _write_failures.clear() 

222 if not keys: 

223 return None 

224 # Key prefix, not the full key: 51 BAs would otherwise make this unreadable. 

225 kinds: dict[str, int] = {} 

226 for k in keys: 

227 kind = k.split(":")[1] if ":" in k else k 

228 kinds[kind] = kinds.get(kind, 0) + 1 

229 return {"count": len(keys), "by_kind": dict(sorted(kinds.items(), key=lambda kv: -kv[1]))}

232def redis_set(key: str, value: dict | list, ttl: int = 86400) -> bool: 

233 """Write a JSON value to Redis with TTL (default 24h). Returns True on success. 

234 

235 **Callers must check the return.** False means the value was NOT written; 

236 treating it as success is how a run reports data it never persisted. 

237 """ 

238 client = _get_redis() 

239 if client is None: 

240 return False 

241 try: 

242 client.setex(key, ttl, json.dumps(value)) 

243 return True 

244 except Exception as exc: 

245 # structlog, not stdlib logging: this needs a `jsonPayload.event` for a 

246 # log-based alert to be able to match it at all. 

247 _log.error("redis_write_failed", key=key, error=str(exc)) 

248 with _write_failure_lock: 

249 _write_failures.append(key) 

250 return False

253class RedisWriteError(RuntimeError): 

254 """A required Redis write failed. Raised by :func:`persist` so a phase that 

255 computed a result but couldn't persist it surfaces as *failed* rather than 

256 silently reporting ok (#268 / P2-03).""" 

257

259def persist(key: str, value: dict | list, ttl: int = 86400) -> None: 

260 """Write to Redis, raising :class:`RedisWriteError` on any failure. 

261 

262 The strict counterpart of :func:`redis_set` (which swallows failures and 

263 returns False). Used by the scoring-job write phases so a dropped write 

264 propagates into the phase's ok-flag — a forecast that computed but couldn't 

265 persist must not count as scored (#268; feeds the #267 region-ok logic). 

266 """ 

267 if not redis_set(key, value, ttl=ttl): 

268 raise RedisWriteError(f"redis write failed for key {key!r}")
